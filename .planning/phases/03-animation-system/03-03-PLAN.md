   ```typescript
   import { MissileType } from '@/combat/CombatSystem';

   export function selectDeathType(killingBlow: MissileType | null): DeathType {
     if (killingBlow === null) {
       // Unknown damage source - random standard death
       return selectRandomStandardDeath();
     }

     switch (killingBlow) {
       case MissileType.FLAME:
       case MissileType.LASER:
         // Fire/laser damage causes melt death
         return DeathType.MELT;

       case MissileType.ROCKET:
       case MissileType.CANNON_SHELL:
       case MissileType.GRENADE:
         // Explosive damage causes missile flip
         return DeathType.MISSILE_FLIP;

       case MissileType.BULLET:
       default:
         // Bullet damage - random standard death (die1-4)
         return selectRandomStandardDeath();
     }
   }

   function selectRandomStandardDeath(): DeathType {
     const deaths = [DeathType.DIE1, DeathType.DIE2, DeathType.DIE3, DeathType.DIE4];
     return deaths[Math.floor(Math.random() * deaths.length)];
   }
   ```

4. **DeathAnimationSystem class**:

   ```typescript
   export class DeathAnimationSystem {
     private scene: Phaser.Scene;
     private activeEffects: DeathEffect[] = [];

     constructor(scene: Phaser.Scene) {
       this.scene = scene;
     }

     /**
      * Spawn a death animation effect at the given position
      */
     spawnRobotDeath(
       x: number,
       y: number,
       team: TeamType,
       killingBlow: MissileType | null
     ): void {
       const deathType = selectDeathType(killingBlow);
       const effect = new DeathEffect(this.scene, x, y, team, deathType);
       this.activeEffects.push(effect);
     }

     /**
      * Update all active death effects
      */
     update(delta: number): void {
       for (let i = this.activeEffects.length - 1; i >= 0; i--) {
         const effect = this.activeEffects[i];
         effect.update(delta);

         if (effect.isComplete()) {
           effect.destroy();
           this.activeEffects.splice(i, 1);
         }
       }
     }
   }
   ```

5. **DeathEffect class** for rendering:
   ```typescript
   class DeathEffect {
     private sprite: Phaser.GameObjects.Sprite;
     private deathType: DeathType;
     private currentFrame: number = 0;
     private frameTimer: number = 0;
     private complete: boolean = false;
     private team: TeamType;

     // Missile flip specific
     private arcX: number = 0;
     private arcY: number = 0;
     private arcTime: number = 0;
     private maxArcTime: number = 3500; // 3.5 seconds in air

     constructor(scene: Phaser.Scene, x: number, y: number, team: TeamType, deathType: DeathType) {
       this.team = team;
       this.deathType = deathType;

       const atlasKey = getRobotAtlasKey(team);
       const frameKey = this.getFrameKey();

       this.sprite = scene.add.sprite(x, y, atlasKey, frameKey);

       if (deathType === DeathType.MISSILE_FLIP) {
         this.initMissileFlip(x, y);
       }
     }

     private initMissileFlip(x: number, y: number): void {
       // Random trajectory from erobotturrent.cpp
       this.arcX = x + (50 - Math.floor(Math.random() * 100));
       this.arcY = y + (50 - Math.floor(Math.random() * 100));
       this.maxArcTime = 3500 + Math.floor(Math.random() * 1000); // 3.5-4.5s
     }

     update(delta: number): void {
       if (this.complete) return;

       this.frameTimer += delta;
       const frameDuration = this.deathType === DeathType.MISSILE_FLIP
         ? (this.arcTime < this.maxArcTime ? 50 : 150) // 50ms in air, 150ms landing
         : ROBOT_ANIMATION_TIMING.DEATH_FRAME_TIME; // 160ms

       if (this.frameTimer >= frameDuration) {
         this.frameTimer = 0;
         this.currentFrame++;

         const maxFrames = DEATH_FRAME_COUNTS[this.deathType];

         // Missile flip loops frames 0-7 while in air, then 8-32 on landing
         if (this.deathType === DeathType.MISSILE_FLIP) {
           this.arcTime += delta;
           if (this.arcTime < this.maxArcTime) {
             if (this.currentFrame > 7) this.currentFrame = 0;
             this.updateArcPosition();
           } else {
             if (this.currentFrame < 8) this.currentFrame = 8;
             if (this.currentFrame >= maxFrames) {
               this.complete = true;
             }
           }
         } else {
           if (this.currentFrame >= maxFrames) {
             this.complete = true;
           }
         }

         // Update sprite frame
         const frameKey = this.getFrameKey();
         this.sprite.setFrame(frameKey);
       }
     }

     private updateArcPosition(): void {
       // Parabolic arc from erobotturrent.cpp
       const progress = this.arcTime / this.maxArcTime;
       const rise = 1.3 + Math.random() * 2; // Random arc height
       const arcHeight = -(rise * progress * progress) + (rise * progress);

       this.sprite.x = this.sprite.x + (this.arcX - this.sprite.x) * 0.05;
       this.sprite.y = this.sprite.y + (this.arcY - this.sprite.y) * 0.05 - arcHeight * 30;
       this.sprite.setScale(1 + arcHeight); // Scale up in air
     }

     private getFrameKey(): string {
       const teamStr = TEAM_STRINGS[this.team];
       const frameStr = this.currentFrame.toString().padStart(2, '0');

       if (this.deathType === DeathType.MELT) {
         return `robot_melt_${teamStr}_n${frameStr}`;
       }

       return `robot_${this.deathType}_${teamStr}_n${frameStr}`;
     }

     isComplete(): boolean {
       return this.complete;
     }

     destroy(): void {
       this.sprite.destroy();
     }
   }
   ```

   ```typescript
   /**
    * Called when robot is destroyed
    * @param killingBlow - The type of attack that killed this unit
    */
   public playDeathAnimation(killingBlow: MissileType | null): void {
     if (this.isDying) return; // Already dying

     this.isDying = true;
     this.killingBlowType = killingBlow;

     // Lock animation state machine
     this.animStateMachine.lock();

     // Determine death type
     const deathType = selectDeathType(killingBlow);
     this.deathType = deathType;

     // Map death type to RobotAnimation enum
     const deathAnimMap: Record<DeathType, RobotAnimation> = {
       [DeathType.DIE1]: RobotAnimation.DIE1,
       [DeathType.DIE2]: RobotAnimation.DIE2,
       [DeathType.DIE3]: RobotAnimation.DIE3,
       [DeathType.DIE4]: RobotAnimation.DIE4,
       [DeathType.MELT]: RobotAnimation.POPE, // pope = melt in original
       [DeathType.MISSILE_FLIP]: RobotAnimation.DIE5,
     };

     this.setAnimation(deathAnimMap[deathType]);

     // Emit death event for effects system
     this.scene.events.emit('robotDeath', {
       x: this.x,
       y: this.y,
       team: this.team,
       killingBlow,
       deathType,
     });
   }
   ```

   ```typescript
   private handleUnitDestroyed(targetRefId: number, attackerRefId: number): void {
     const target = this.objectManager.getObject(targetRefId);
     if (!target) return;

     // Determine killing blow type from attacker
     const attacker = this.objectManager.getObject(attackerRefId);
     const killingBlow = this.getKillingBlowType(attacker);

     if (target instanceof Robot) {
       // For standard deaths, robot renders itself
       target.playDeathAnimation(killingBlow);

       // For missile flip, spawn the arc effect
       if (killingBlow === MissileType.ROCKET ||
           killingBlow === MissileType.CANNON_SHELL ||
           killingBlow === MissileType.GRENADE) {
         this.deathAnimationSystem.spawnRobotDeath(
           target.x,
           target.y,
           target.team,
           killingBlow
         );
       }
     }

     // Note: Don't remove unit yet - wait for death animation to complete
   }

   private getKillingBlowType(attacker: GameObject | null): MissileType | null {
     if (!attacker) return null;

     // Determine missile type from attacker
     // This logic should match CombatSystem.getMissileType()
     if (attacker instanceof Robot) {
       const robotType = attacker.getRobotType();
       switch (robotType) {
         case RobotType.PYRO: return MissileType.FLAME;
         case RobotType.LASER: return MissileType.LASER;
         case RobotType.TOUGH: return MissileType.ROCKET;
         default: return MissileType.BULLET;
       }
     }
     // ... similar for vehicles and cannons
     return MissileType.BULLET;
   }
   ```
