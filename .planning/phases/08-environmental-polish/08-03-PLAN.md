/**
 * TrackEffect - Vehicle track marks on terrain
 * Ported from source/etrack.cpp
 *
 * Track marks fade over ~4 seconds with 3 fade stages:
 * - 0-3.3s: Full opacity (ti=0)
 * - 3.3-3.6s: Medium fade (ti=1)
 * - 3.6-3.9s: Fading out (ti=2)
 * - 3.9s+: Removed
 */
import Phaser from "phaser";

export enum TrackType {
  TANK = 0,
  JEEP = 1,
}

// Track dimensions and colors
const TRACK_CONFIG = {
  // Track dimensions
  width: 3,
  length: 8,

  // Colors by type
  colors: {
    [TrackType.TANK]: 0x4a3a2a, // Dark brown
    [TrackType.JEEP]: 0x3a3a3a, // Dark gray
  },

  // Fade timing (from etrack.cpp)
  fadeTime1: 3300, // ms - start first fade
  fadeTime2: 3600, // ms - second fade
  fadeTime3: 3900, // ms - remove

  // Alpha values for each stage
  alphas: [0.6, 0.4, 0.2],
};

export class TrackEffect extends Phaser.GameObjects.Container {
  private trackType: TrackType;
  private graphics: Phaser.GameObjects.Graphics;

  // Timing
  private startTime: number;
  private currentStage: number = 0;
  public killMe: boolean = false;

  // Track positions (two marks for left/right treads)
  private trackPositions: Array<{ x: number; y: number }> = [];

  constructor(
    scene: Phaser.Scene,
    centerX: number,
    centerY: number,
    direction: number, // 0-7 for 8 directions
    trackType: TrackType,
    layLeft: boolean = true,
    layRight: boolean = true
  ) {
    super(scene, 0, 0);

    this.trackType = trackType;
    this.startTime = scene.time.now + Math.random() * 100; // Slight random offset

    // Create graphics
    this.graphics = scene.add.graphics();
    this.add(this.graphics);

    // Calculate track positions based on direction
    this.calculateTrackPositions(centerX, centerY, direction, layLeft, layRight);

    // Draw initial tracks
    this.drawTracks(TRACK_CONFIG.alphas[0] ?? 0.6);

    // Set depth below units
    this.setDepth(2);

    scene.add.existing(this);
  }

  /**
   * Calculate track positions relative to vehicle center
   * Based on etrack.cpp SetTrackCoords
   */
  private calculateTrackPositions(
    cx: number,
    cy: number,
    direction: number,
    layLeft: boolean,
    layRight: boolean
  ): void {
    let x1 = 0, y1 = 0, x2 = 0, y2 = 0;

    // Track offsets based on direction (from etrack.cpp)
    switch (direction) {
      case 0: // East (0 degrees)
        x1 = x2 = cx - 15;
        y1 = cy - 6 + 4;
        y2 = cy + 6 + 4;
        break;
      case 4: // West (180 degrees)
        x1 = x2 = cx + 17;
        y1 = cy - 6 + 4;
        y2 = cy + 6 + 4;
        break;
      case 2: // South (90 degrees)
        y1 = y2 = cy + 15;
        x1 = cx - 8;
        x2 = cx + 8;
        break;
      case 6: // North (270 degrees)
        y1 = y2 = cy - 11;
        x1 = cx - 8;
        x2 = cx + 8;
        break;
      case 1: // SE (45 degrees)
        x1 = cx - 14;
        y1 = cy + 4 + 3;
        x2 = cx + 4;
        y2 = cy + 14 + 3;
        break;
      case 5: // NW (225 degrees)
        x1 = cx + 14 - 1;
        y1 = cy - 4 + 4;
        x2 = cx - 4 - 1;
        y2 = cy - 14 + 4;
        break;
      case 3: // SW (135 degrees)
        x1 = cx + 14 + 1;
        y1 = cy + 4 + 4;
        x2 = cx + 4 + 1;
        y2 = cy + 14 + 4;
        break;
      case 7: // NE (315 degrees)
        x1 = cx - 14;
        y1 = cy - 4 + 3;
        x2 = cx - 4;
        y2 = cy - 14 + 3;
        break;
      default:
        x1 = x2 = cx;
        y1 = y2 = cy;
    }

    // Add small random offset (from etrack.cpp)
    const rx = Math.floor(Math.random() * 2);
    const ry = Math.floor(Math.random() * 2);
    x1 += rx;
    x2 += rx;
    y1 += ry;
    y2 += ry;

    if (layLeft) {
      this.trackPositions.push({ x: x1, y: y1 });
    }
    if (layRight) {
      this.trackPositions.push({ x: x2, y: y2 });
    }
  }

  /**
   * Draw track marks with given alpha
   */
  private drawTracks(alpha: number): void {
    this.graphics.clear();

    const color = TRACK_CONFIG.colors[this.trackType] ?? TRACK_CONFIG.colors[TrackType.TANK];

    for (const pos of this.trackPositions) {
      this.graphics.fillStyle(color ?? 0x4a3a2a, alpha);

      // Draw small rectangle for track mark
      this.graphics.fillRect(
        pos.x - TRACK_CONFIG.width / 2,
        pos.y - TRACK_CONFIG.length / 2,
        TRACK_CONFIG.width,
        TRACK_CONFIG.length
      );
    }
  }

  /**
   * Update track effect - handle fade stages
   */
  public update(time: number, _delta: number): void {
    if (this.killMe) return;

    const elapsed = time - this.startTime;

    // Check fade stages
    if (elapsed >= TRACK_CONFIG.fadeTime3) {
      this.killMe = true;
      return;
    }

    let newStage: number;
    if (elapsed >= TRACK_CONFIG.fadeTime2) {
      newStage = 2;
    } else if (elapsed >= TRACK_CONFIG.fadeTime1) {
      newStage = 1;
    } else {
      newStage = 0;
    }

    // Redraw if stage changed
    if (newStage !== this.currentStage) {
      this.currentStage = newStage;
      this.drawTracks(TRACK_CONFIG.alphas[newStage] ?? 0.2);
    }
  }

  public override destroy(): void {
    this.graphics.destroy();
    super.destroy();
  }
}
```

/**
 * Create vehicle track effect
 * Called by vehicles as they move
 *
 * @param centerX Vehicle center X position
 * @param centerY Vehicle center Y position
 * @param direction Movement direction (0-7)
 * @param trackType Type of track (TANK or JEEP)
 * @param layLeft Whether to lay left track
 * @param layRight Whether to lay right track
 */
public createTrackEffect(
  centerX: number,
  centerY: number,
  direction: number,
  trackType: TrackType,
  layLeft: boolean = true,
  layRight: boolean = true
): void {
  // Limit total track effects for performance
  if (this.trackEffects.length >= EffectsSystem.MAX_TRACK_EFFECTS) {
    // Remove oldest track
    const oldest = this.trackEffects.shift();
    if (oldest) {
      oldest.destroy();
    }
  }

  const track = new TrackEffect(
    this.scene,
    centerX,
    centerY,
    direction,
    trackType,
    layLeft,
    layRight
  );

  this.trackEffects.push(track);
}
```

 * Drop track marks while moving

protected tryDropTracks(time: number): void {
  if (!this.effectsSystem) return;

  // Check if enough time has passed
  if (time - this.lastTrackDropTime < this.trackDropInterval) {
    return;
  }
  this.lastTrackDropTime = time;

  // Get current direction (0-7)
  const direction = this.getMovementDirection();

  // Drop tracks
  this.effectsSystem.createTrackEffect(
    this.x,
    this.y,
    direction,
    this.trackType,
    true, // left track
    true  // right track
  );
}
