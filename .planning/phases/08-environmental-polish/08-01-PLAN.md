/**
 * HutAnimal - Animals that spawn from huts and wander
 * Ported from source/ahutanimal.cpp
 */
import Phaser from "phaser";
import { AnimalType, PlanetType } from "@/types";

export enum AnimalState {
  NOTHING = 0,
  WALKING = 1,
  LOOKING = 2,
}

// Animal colors for placeholder rendering (matches original sprites)
const ANIMAL_COLORS: Record<AnimalType, number> = {
  [AnimalType.GREEN_SNAKE]: 0x228b22,
  [AnimalType.GREEN_LIZARD]: 0x32cd32,
  [AnimalType.DESERT_RABBIT]: 0xd2b48c,
  [AnimalType.RAPTOR]: 0x8b4513,
  [AnimalType.MINI_RAPTOR]: 0xa0522d,
  [AnimalType.PIG_DINO]: 0xffc0cb,
  [AnimalType.YELLOW_WORM]: 0xffd700,
  [AnimalType.ARCTIC_RABBIT]: 0xfffafa,
  [AnimalType.PENGUIN]: 0x000000,
  [AnimalType.WHITE_WOLF]: 0xf5f5f5,
  [AnimalType.OSTRICH]: 0x8b8989,
  [AnimalType.RAT]: 0x696969,
  [AnimalType.TURTLE]: 0x556b2f,
  [AnimalType.RED_WORM]: 0xff6347,
  [AnimalType.GREEN_EYED_FOX]: 0xd2691e,
};

// Palette to animal type mapping (from ahutanimal.cpp)
const ANIMALS_BY_PALETTE: Record<PlanetType, AnimalType[]> = {
  [PlanetType.DESERT]: [AnimalType.GREEN_SNAKE, AnimalType.GREEN_LIZARD, AnimalType.DESERT_RABBIT],
  [PlanetType.VOLCANIC]: [AnimalType.RAPTOR, AnimalType.MINI_RAPTOR, AnimalType.PIG_DINO, AnimalType.YELLOW_WORM],
  [PlanetType.ARCTIC]: [AnimalType.ARCTIC_RABBIT, AnimalType.PENGUIN, AnimalType.WHITE_WOLF],
  [PlanetType.JUNGLE]: [AnimalType.OSTRICH, AnimalType.RAT, AnimalType.TURTLE],
  [PlanetType.CITY]: [AnimalType.RED_WORM, AnimalType.RAT, AnimalType.GREEN_EYED_FOX],
};

export class HutAnimal extends Phaser.GameObjects.Container {
  // Identity
  private animalType: AnimalType;
  private palette: PlanetType;

  // Home reference
  private homeX: number;
  private homeY: number;
  private roamDistance: number = 128; // pixels

  // State
  private state: AnimalState = AnimalState.NOTHING;
  private goingHome: boolean = false;
  public killMe: boolean = false;

  // Movement - from ahutanimal.cpp
  private moveSpeed: number = 15; // pixels/sec
  private targetX: number = 0;
  private targetY: number = 0;
  private dx: number = 0;
  private dy: number = 0;
  private direction: number = 0; // 0-7 for 8 directions

  // Animation timing - from ahutanimal.cpp
  private walkFrame: number = 0;
  private walkTimer: number = 0;
  private walkSpeed: number = 200; // ms per frame
  private lookFrame: number = 0;
  private lookTimer: number = 0;
  private lookSpeed: number = 350; // ms per frame

  // State timing
  private nextStateTime: number = 0;

  // Visuals
  private graphics: Phaser.GameObjects.Graphics;

  constructor(
    scene: Phaser.Scene,
    x: number,
    y: number,
    homeX: number,
    homeY: number,
    palette: PlanetType
  ) {
    super(scene, x, y);

    this.homeX = homeX;
    this.homeY = homeY;
    this.palette = palette;
    this.animalType = this.chooseRandomAnimal(palette);

    // Create placeholder graphics
    this.graphics = scene.add.graphics();
    this.add(this.graphics);
    this.drawAnimal();

    this.setDepth(5); // Just above ground
    scene.add.existing(this);

    // Start with nothing state, will transition to walking
    this.setStateNothing();
  }

  private chooseRandomAnimal(palette: PlanetType): AnimalType {
    const options = ANIMALS_BY_PALETTE[palette] ?? ANIMALS_BY_PALETTE[PlanetType.DESERT];
    return options[Math.floor(Math.random() * options.length)] ?? AnimalType.GREEN_SNAKE;
  }

  private drawAnimal(): void {
    this.graphics.clear();
    const color = ANIMAL_COLORS[this.animalType] ?? 0x888888;
    const size = 4 + (this.animalType % 3); // Vary size slightly

    // Simple circle placeholder
    this.graphics.fillStyle(color, 1);
    this.graphics.fillCircle(0, 0, size);

    // Direction indicator (small line in movement direction)
    if (this.state === AnimalState.WALKING) {
      const angle = (this.direction * Math.PI) / 4;
      this.graphics.lineStyle(1, color, 0.8);
      this.graphics.lineBetween(0, 0, Math.cos(angle) * size * 1.5, Math.sin(angle) * size * 1.5);
    }
  }

  /**
   * Update animal each frame
   */
  public update(time: number, delta: number): void {
    if (this.killMe) return;

    // Process movement
    if (this.isMoving()) {
      this.processMovement(delta);
    }

    // Process state
    switch (this.state) {
      case AnimalState.NOTHING:
        if (time >= this.nextStateTime) {
          // 80% chance to walk, 20% to look
          if (Math.random() > 0.2) {
            this.gotoRandomTile();
          } else {
            this.setStateLooking();
          }
        }
        break;

      case AnimalState.LOOKING:
        this.lookTimer += delta;
        if (this.lookTimer >= this.lookSpeed) {
          this.lookTimer = 0;
          this.lookFrame++;
          if (this.lookFrame >= 4) {
            // Done looking, back to nothing
            if (Math.random() > 0.2) {
              this.setStateNothing();
            } else {
              this.lookFrame = 0; // Look again
            }
          }
        }
        break;

      case AnimalState.WALKING:
        this.walkTimer += delta;
        if (this.walkTimer >= this.walkSpeed) {
          this.walkTimer = 0;
          this.walkFrame = (this.walkFrame + 1) % 4;
          this.drawAnimal();
        }
        break;
    }
  }

  private isMoving(): boolean {
    return this.state === AnimalState.WALKING && (this.dx !== 0 || this.dy !== 0);
  }

  private processMovement(delta: number): void {
    const dt = delta / 1000;
    this.x += this.dx * dt;
    this.y += this.dy * dt;

    // Check if reached target
    const distX = this.targetX - this.x;
    const distY = this.targetY - this.y;
    const dist = Math.sqrt(distX * distX + distY * distY);

    if (dist < 4) {
      // Reached target
      this.x = this.targetX;
      this.y = this.targetY;

      if (this.goingHome && Math.abs(this.x - this.homeX) < 8 && Math.abs(this.y - this.homeY) < 8) {
        // Home - kill this animal
        this.killMe = true;
        return;
      }

      // 66% chance to keep walking, 33% to stop
      if (Math.random() > 0.33) {
        this.gotoRandomTile();
      } else {
        this.setStateNothing();
        this.dx = 0;
        this.dy = 0;
      }
    }
  }

  private setStateNothing(): void {
    this.state = AnimalState.NOTHING;
    this.nextStateTime = this.scene.time.now + 100 + Math.random() * 1000;
    this.walkFrame = 0;
    this.drawAnimal();
  }

  private setStateLooking(): void {
    this.state = AnimalState.LOOKING;
    this.lookFrame = 0;
    this.lookTimer = 0;
  }

  private gotoRandomTile(): void {
    // Pick a random adjacent position
    const offsetX = (Math.random() - 0.5) * 32;
    const offsetY = (Math.random() - 0.5) * 32;

    let targetX = this.x + offsetX;
    let targetY = this.y + offsetY;

    // Check roam distance from home
    const distFromHome = Math.sqrt(
      (targetX - this.homeX) ** 2 + (targetY - this.homeY) ** 2
    );

    if (distFromHome > this.roamDistance) {
      // Too far, head back toward home
      const angle = Math.atan2(this.homeY - this.y, this.homeX - this.x);
      targetX = this.x + Math.cos(angle) * 16;
      targetY = this.y + Math.sin(angle) * 16;
    }

    this.gotoPosition(targetX, targetY);
  }

  private gotoPosition(tx: number, ty: number): void {
    this.targetX = tx;
    this.targetY = ty;

    const distX = tx - this.x;
    const distY = ty - this.y;
    const dist = Math.sqrt(distX * distX + distY * distY);

    if (dist < 1) {
      this.setStateNothing();
      return;
    }

    this.dx = (distX / dist) * this.moveSpeed;
    this.dy = (distY / dist) * this.moveSpeed;

    // Calculate direction (0-7)
    const angle = Math.atan2(this.dy, this.dx);
    this.direction = Math.round(((angle + Math.PI) / Math.PI) * 4) % 8;

    this.state = AnimalState.WALKING;
    this.drawAnimal();
  }

  /**
   * Send animal back to home hut (will be killed when arrives)
   */
  public goHome(): void {
    this.goingHome = true;
    this.gotoPosition(this.homeX, this.homeY);
  }

  public isGoingHome(): boolean {
    return this.goingHome;
  }

  public override destroy(): void {
    this.graphics.destroy();
    super.destroy();
  }
}

/**
 * Hut - Spawns and manages animals
 * Ported from source/ohut.cpp
 */
import Phaser from "phaser";
import { ObjectType, ItemType, PlanetType } from "@/types";
import type { ItemState } from "@/types";
import { GameObject } from "../GameObject";
import { HutAnimal } from "../animals/HutAnimal";

// Configuration from zsettings
const HUT_CONFIG = {
  animalMin: 1,
  animalMax: 5,
  roamDistance: 128,
  checkInterval: 1000, // ms between animal count checks
  maxIntervalChange: 10000, // ms between max animal changes
};

export class Hut extends GameObject {
  private palette: PlanetType = PlanetType.DESERT;
  private animals: HutAnimal[] = [];
  private maxAnimals: number = 0;

  // Timing
  private nextAnimalCheckTime: number = 0;
  private nextMaxChangeTime: number = 0;

  // Graphics
  private hutGraphics: Phaser.GameObjects.Graphics;

  constructor(scene: Phaser.Scene, state: ItemState) {
    super(scene, state);

    // Determine palette from map (default to DESERT)
    // In real implementation, this would come from map data
    this.palette = PlanetType.DESERT;

    // Create placeholder hut graphic
    this.hutGraphics = scene.add.graphics();
    this.add(this.hutGraphics);
    this.drawHut();

    this.setDepth(10);
    this.setMaxAnimals();

    // Start checking for animals after a short delay
    this.nextAnimalCheckTime = scene.time.now + 2000;
  }

  private drawHut(): void {
    this.hutGraphics.clear();

    // Simple hut placeholder - brown rectangle with triangle roof
    this.hutGraphics.fillStyle(0x8b4513, 1);
    this.hutGraphics.fillRect(-8, -4, 16, 12);

    // Roof
    this.hutGraphics.fillStyle(0x654321, 1);
    this.hutGraphics.fillTriangle(-10, -4, 10, -4, 0, -12);

    // Door
    this.hutGraphics.fillStyle(0x2f1810, 1);
    this.hutGraphics.fillRect(-2, 2, 4, 6);
  }

  /**
   * Update hut and manage animals
   */
  public override update(time: number, delta: number): void {
    super.update(time, delta);

    // Update all animals
    for (let i = this.animals.length - 1; i >= 0; i--) {
      const animal = this.animals[i];
      if (!animal) continue;

      if (animal.killMe) {
        animal.destroy();
        this.animals.splice(i, 1);
      } else {
        animal.update(time, delta);
      }
    }

    // Check animal count periodically
    if (time >= this.nextAnimalCheckTime) {
      this.nextAnimalCheckTime = time + HUT_CONFIG.checkInterval;
      this.manageAnimals();
    }

    // Change max animals periodically
    if (time >= this.nextMaxChangeTime) {
      this.nextMaxChangeTime = time + HUT_CONFIG.maxIntervalChange;
      this.setMaxAnimals();
    }
  }

  /**
   * Spawn or despawn animals to match max count
   */
  private manageAnimals(): void {
    const displacement = this.maxAnimals - this.animals.length;

    if (displacement > 0) {
      // Need to spawn animals
      const spawnCount = Math.floor(Math.random() * (displacement + 1));
      this.createAnimals(spawnCount);
    } else if (displacement < 0) {
      // Need to send some home
      const sendHomeCount = Math.floor(Math.random() * (-displacement + 1));
      this.sendAnimalsHome(sendHomeCount);
    }
  }

  /**
   * Create new animals at exit tiles
   */
  private createAnimals(count: number): void {
    for (let i = 0; i < count; i++) {
      const exitPos = this.getExitPosition();
      if (exitPos) {
        const animal = new HutAnimal(
          this.scene,
          this.x,
          this.y,
          this.x,
          this.y,
          this.palette
        );

        // Move to exit position
        animal.setPosition(this.x, this.y);
        this.animals.push(animal);
      }
    }
  }

  /**
   * Get a valid exit position near the hut
   */
  private getExitPosition(): { x: number; y: number } | null {
    // Default exit is below the hut
    const exitX = this.x;
    const exitY = this.y + 16;

    // TODO: Check passability when map pathfinding is accessible
    return { x: exitX, y: exitY };
  }

  /**
   * Send animals back home
   */
  private sendAnimalsHome(count: number): void {
    let sentCount = 0;

    for (const animal of this.animals) {
      if (!animal.isGoingHome()) {
        animal.goHome();
        sentCount++;
        if (sentCount >= count) break;
      }
    }
  }

  /**
   * Set random max animals
   */
  private setMaxAnimals(): void {
    const range = HUT_CONFIG.animalMax - HUT_CONFIG.animalMin;
    this.maxAnimals = HUT_CONFIG.animalMin + Math.floor(Math.random() * (range + 1));
  }

  /**
   * Set terrain palette for animal type selection
   */
  public setPalette(palette: PlanetType): void {
    this.palette = palette;
  }

  public override destroy(): void {
    // Destroy all animals
    for (const animal of this.animals) {
      animal.destroy();
    }
    this.animals = [];
    this.hutGraphics.destroy();
    super.destroy();
  }
}

export default Hut;
