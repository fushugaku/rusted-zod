/**
 * VisibilitySystem - Tracks fog of war and tile visibility
 *
 * Tile states:
 * - UNEXPLORED: Never seen (black fog)
 * - EXPLORED: Seen before but not visible now (dim)
 * - VISIBLE: Currently visible (no fog)
 */
import { TeamType, ObjectType } from "@/types";
import type { GameObject } from "@/objects/GameObject";

export enum TileVisibility {
  UNEXPLORED = 0,
  EXPLORED = 1,
  VISIBLE = 2,
}

// Default sight range in tiles (can be overridden per unit)
const DEFAULT_SIGHT_RANGE = 8;

// Sight ranges by unit type (in tiles)
const SIGHT_RANGES: Partial<Record<number, number>> = {
  // Robots have moderate sight
  // Vehicles have better sight
  // Buildings have limited sight
};

export class VisibilitySystem {
  // Visibility grid - one entry per tile
  private visibility: TileVisibility[] = [];
  private mapWidth: number = 0;
  private mapHeight: number = 0;
  private tileWidth: number = 16;
  private tileHeight: number = 16;

  // Player team for visibility calculations
  private playerTeam: TeamType = TeamType.RED;

  // Unit lookup for visibility sources
  private getUnits: (() => GameObject[]) | null = null;

  // Cached visible set for current frame (optimization)
  private currentVisibleTiles: Set<number> = new Set();

  // Update throttling
  private lastUpdateTime: number = 0;
  private updateInterval: number = 100; // ms

  constructor() {}

  /**
   * Initialize visibility grid for map
   */
  public initialize(
    mapWidth: number,
    mapHeight: number,
    tileWidth: number = 16,
    tileHeight: number = 16
  ): void {
    this.mapWidth = mapWidth;
    this.mapHeight = mapHeight;
    this.tileWidth = tileWidth;
    this.tileHeight = tileHeight;

    // Create visibility grid - all unexplored initially
    const totalTiles = mapWidth * mapHeight;
    this.visibility = new Array(totalTiles).fill(TileVisibility.UNEXPLORED);
  }

  /**
   * Set player team for visibility calculations
   */
  public setPlayerTeam(team: TeamType): void {
    this.playerTeam = team;
  }

  /**
   * Set unit lookup callback
   */
  public setUnitLookup(getUnits: () => GameObject[]): void {
    this.getUnits = getUnits;
  }

  /**
   * Update visibility based on unit positions
   */
  public update(time: number): void {
    // Throttle updates
    if (time - this.lastUpdateTime < this.updateInterval) {
      return;
    }
    this.lastUpdateTime = time;

    if (!this.getUnits) return;

    // Clear current visible set
    this.currentVisibleTiles.clear();

    // Get all player units
    const units = this.getUnits();

    for (const unit of units) {
      // Only player team provides visibility
      if (unit.getTeam() !== this.playerTeam) continue;

      // Get unit position in tiles
      const tileX = Math.floor(unit.x / this.tileWidth);
      const tileY = Math.floor(unit.y / this.tileHeight);

      // Get sight range for this unit type
      const sightRange = this.getSightRange(unit);

      // Mark tiles within sight range as visible
      this.revealTilesInRadius(tileX, tileY, sightRange);
    }

    // Update visibility grid: VISIBLE -> EXPLORED for tiles no longer visible
    for (let i = 0; i < this.visibility.length; i++) {
      if (this.currentVisibleTiles.has(i)) {
        this.visibility[i] = TileVisibility.VISIBLE;
      } else if (this.visibility[i] === TileVisibility.VISIBLE) {
        // Was visible, now only explored
        this.visibility[i] = TileVisibility.EXPLORED;
      }
    }
  }

  /**
   * Reveal tiles within radius of a point
   */
  private revealTilesInRadius(centerX: number, centerY: number, radius: number): void {
    const radiusSq = radius * radius;

    const minX = Math.max(0, centerX - radius);
    const maxX = Math.min(this.mapWidth - 1, centerX + radius);
    const minY = Math.max(0, centerY - radius);
    const maxY = Math.min(this.mapHeight - 1, centerY + radius);

    for (let y = minY; y <= maxY; y++) {
      for (let x = minX; x <= maxX; x++) {
        const dx = x - centerX;
        const dy = y - centerY;
        const distSq = dx * dx + dy * dy;

        if (distSq <= radiusSq) {
          const index = y * this.mapWidth + x;
          this.currentVisibleTiles.add(index);
        }
      }
    }
  }

  /**
   * Get sight range for a unit
   */
  private getSightRange(unit: GameObject): number {
    // Use attack range as base, with minimum
    // In real implementation, could vary by unit type
    const baseRange = DEFAULT_SIGHT_RANGE;

    // Buildings have smaller sight range
    if (unit.getObjectType() === ObjectType.OBJECT_BUILDING) {
      return Math.max(4, baseRange - 2);
    }

    return baseRange;
  }

  /**
   * Get visibility state for a tile
   */
  public getTileVisibility(tileX: number, tileY: number): TileVisibility {
    if (tileX < 0 || tileX >= this.mapWidth || tileY < 0 || tileY >= this.mapHeight) {
      return TileVisibility.UNEXPLORED;
    }

    const index = tileY * this.mapWidth + tileX;
    return this.visibility[index] ?? TileVisibility.UNEXPLORED;
  }

  /**
   * Get visibility for a world position
   */
  public getVisibilityAtPosition(x: number, y: number): TileVisibility {
    const tileX = Math.floor(x / this.tileWidth);
    const tileY = Math.floor(y / this.tileHeight);
    return this.getTileVisibility(tileX, tileY);
  }

  /**
   * Check if a position is currently visible
   */
  public isVisible(x: number, y: number): boolean {
    return this.getVisibilityAtPosition(x, y) === TileVisibility.VISIBLE;
  }

  /**
   * Check if a position has been explored
   */
  public isExplored(x: number, y: number): boolean {
    const visibility = this.getVisibilityAtPosition(x, y);
    return visibility === TileVisibility.EXPLORED || visibility === TileVisibility.VISIBLE;
  }

  /**
   * Get map dimensions
   */
  public getMapSize(): { width: number; height: number } {
    return { width: this.mapWidth, height: this.mapHeight };
  }

  /**
   * Get raw visibility grid for minimap rendering
   */
  public getVisibilityGrid(): TileVisibility[] {
    return this.visibility;
  }

  /**
   * Reveal entire map (for debug/cheat)
   */
  public revealAll(): void {
    this.visibility.fill(TileVisibility.VISIBLE);
  }

  /**
   * Reset visibility to unexplored
   */
  public reset(): void {
    this.visibility.fill(TileVisibility.UNEXPLORED);
    this.currentVisibleTiles.clear();
  }
}

export default VisibilitySystem;
```

/**
 * Draw fog of war overlay
 */
private drawFog(): void {
  this.fogLayer.clear();

  if (!this.visibilitySystem) return;

  const { width: mapTileWidth, height: mapTileHeight } = this.visibilitySystem.getMapSize();
  if (mapTileWidth === 0 || mapTileHeight === 0) return;

  const visibilityGrid = this.visibilitySystem.getVisibilityGrid();

  // Calculate pixels per tile on minimap
  const pixelsPerTileX = this.minimapWidth / mapTileWidth;
  const pixelsPerTileY = this.minimapHeight / mapTileHeight;

  // Draw fog for each tile
  for (let ty = 0; ty < mapTileHeight; ty++) {
    for (let tx = 0; tx < mapTileWidth; tx++) {
      const index = ty * mapTileWidth + tx;
      const visibility = visibilityGrid[index] ?? TileVisibility.UNEXPLORED;

      // Skip fully visible tiles (no fog needed)
      if (visibility === TileVisibility.VISIBLE) continue;

      // Calculate position on minimap
      const px = tx * pixelsPerTileX;
      const py = ty * pixelsPerTileY;

      // Draw fog with appropriate alpha
      if (visibility === TileVisibility.UNEXPLORED) {
        // Dark fog for unexplored
        this.fogLayer.fillStyle(0x000000, 0.85);
      } else {
        // Lighter fog for explored but not visible
        this.fogLayer.fillStyle(0x000000, 0.45);
      }

      this.fogLayer.fillRect(px, py, Math.ceil(pixelsPerTileX), Math.ceil(pixelsPerTileY));
    }
  }
}
```
