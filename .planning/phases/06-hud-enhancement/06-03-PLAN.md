public updateFromSelection(states: GameObjectState[]): void {
  if (states.length === 0) {
    this.setVisible(false);
    return;
  }

  // Single selection - hide group panel (use unit panel instead)
  if (states.length === 1) {
    this.setVisible(false);
    return;
  }

  this.setVisible(true);

  // Count units by type
  const unitCounts = this.aggregateUnits(states);

  // Update total count
  this.totalCountText.setText(`${states.length} Units`);

  // Calculate total health
  let totalHealth = 0;
  let totalMaxHealth = 0;
  for (const state of states) {
    totalHealth += state.health;
    totalMaxHealth += state.maxHealth;
  }

  // Update health bar
  this.drawHealthBar(totalHealth, totalMaxHealth);

  // Update unit list
  this.updateUnitList(unitCounts);
}

private aggregateUnits(states: GameObjectState[]): UnitCount[] {
  const counts = new Map<string, UnitCount>();

  for (const state of states) {
    const typeName = this.getUnitTypeName(state);
    const existing = counts.get(typeName);

    if (existing) {
      existing.count++;
      existing.totalHealth += state.health;
      existing.maxHealth += state.maxHealth;
    } else {
      counts.set(typeName, {
        type: typeName,
        count: 1,
        totalHealth: state.health,
        maxHealth: state.maxHealth,
      });
    }
  }

  // Sort by count descending
  return Array.from(counts.values()).sort((a, b) => b.count - a.count);
}

private getUnitTypeName(state: GameObjectState): string {
  switch (state.objectType) {
    case ObjectType.OBJECT_ROBOT:
      return RobotType[(state as RobotState).robotType] ?? "Robot";
    case ObjectType.OBJECT_VEHICLE:
      return VehicleType[(state as VehicleState).vehicleType]?.replace(/_/g, " ") ?? "Vehicle";
    case ObjectType.OBJECT_CANNON:
      return CannonType[(state as CannonState).cannonType]?.replace(/_/g, " ") ?? "Cannon";
    default:
      return "Unknown";
  }
}

private drawHealthBar(current: number, max: number): void {
  this.healthBarBg.clear();
  this.healthBarFill.clear();

  const barWidth = PANEL_WIDTH - 16;
  const barHeight = 8;
  const percent = max > 0 ? current / max : 0;

  // Background
  this.healthBarBg.fillStyle(0x333333, 1);
  this.healthBarBg.fillRect(0, 0, barWidth, barHeight);

  // Fill with color based on percentage
  let color = COLORS.healthGood;
  if (percent < 0.3) color = COLORS.healthLow;
  else if (percent < 0.6) color = COLORS.healthMedium;

  this.healthBarFill.fillStyle(color, 1);
  this.healthBarFill.fillRect(0, 0, barWidth * percent, barHeight);

  // Text
  const healthPercent = Math.round(percent * 100);
  this.healthText.setText(`${healthPercent}% Health`);
}

private updateUnitList(counts: UnitCount[]): void {
  // Clear existing rows
  for (const row of this.unitRows) {
    row.destroy();
  }
  this.unitRows = [];

  // Create new rows (max 5 shown)
  const maxRows = Math.min(counts.length, 5);
  for (let i = 0; i < maxRows; i++) {
    const count = counts[i];
    const y = i * ROW_HEIGHT;

    const row = this.scene.add.text(0, y, `${count.count}x ${count.type}`, {
      fontFamily: "Courier New",
      fontSize: "9px",
      color: "#aaaaaa",
    });
    this.unitListContainer.add(row);
    this.unitRows.push(row);
  }

  // Show overflow indicator
  if (counts.length > 5) {
    const overflow = this.scene.add.text(0, 5 * ROW_HEIGHT, `+${counts.length - 5} more...`, {
      fontFamily: "Courier New",
      fontSize: "9px",
      color: "#666666",
    });
    this.unitListContainer.add(overflow);
    this.unitRows.push(overflow);
  }

private updateQueueDisplay(production: BuildingProduction): void {
  // Progress bar
  this.progressBar.clear();

  const barWidth = PANEL_WIDTH - 16;
  const barHeight = 8;

  this.progressBar.fillStyle(COLORS.progressBg, 1);
  this.progressBar.fillRect(0, 0, barWidth, barHeight);

  if (production.currentUnit) {
    const progress = production.buildProgress / production.buildTime;
    this.progressBar.fillStyle(COLORS.progressFill, 1);
    this.progressBar.fillRect(0, 0, barWidth * progress, barHeight);

    const remaining = Math.ceil((production.buildTime - production.buildProgress) / 1000);
    const mins = Math.floor(remaining / 60);
    const secs = remaining % 60;
    this.progressText.setText(`${mins}:${secs.toString().padStart(2, '0')}`);
  } else {
    this.progressText.setText("Idle");
  }

  // Queue slots
  const queue = production.currentUnit
    ? [production.currentUnit, ...production.queue]
    : production.queue;

  for (let i = 0; i < MAX_QUEUE_DISPLAY; i++) {
    const slot = this.queueSlots[i];
    const icon = this.queueIcons[i];

    slot.clear();

    if (i < queue.length) {

      // Draw filled slot

/**
 * Update HUD panels based on current selection
 * @param states Array of selected unit states
 * @param primaryState The primary selected unit (for single selection display)
 */
public updateSelection(states: GameObjectState[], primaryState: GameObjectState | null): void {
  // Hide all panels first
  this.unitInfoPanel?.setVisible(false);
  this.groupInfoPanel?.setVisible(false);
  this.buildingInfoPanel?.setVisible(false);
  this.portrait?.setVisible(false);

  if (states.length === 0) {
    // Nothing selected
    return;
  }

  if (states.length === 1 && primaryState) {
    // Single selection
    if (primaryState.objectType === ObjectType.OBJECT_BUILDING) {
      // Show building panel
      const buildingState = primaryState as BuildingObjectState;
      const production = this.getProductionForBuilding(buildingState.refId);
      const zoneOwnership = this.getZoneOwnership(buildingState.zoneId);
      this.buildingInfoPanel.updateFromState(buildingState, production, zoneOwnership);
      // Portrait for buildings could show construction icon or be hidden
    } else {
      // Show unit info panel + portrait
      this.unitInfoPanel?.updateFromState(primaryState);
      this.updatePortrait(primaryState);
    }
  } else {
    // Multi-selection
    this.groupInfoPanel.updateFromSelection(states);

    // Show portrait of primary unit if available
    if (primaryState && primaryState.objectType !== ObjectType.OBJECT_BUILDING) {
      this.updatePortrait(primaryState);
    }
  }
}

private updatePortrait(state: GameObjectState): void {
  if (!this.portrait) return;

  if (state.objectType === ObjectType.OBJECT_ROBOT) {
    const robotState = state as RobotState;
    this.portrait.setRobotType(robotState.robotType, state.team);
    this.portrait.setInVehicle(false);
    this.portrait.setVisible(true);
    // Don't auto-play reporting anim on multi-select
  } else if (state.objectType === ObjectType.OBJECT_VEHICLE) {
    const vehicleState = state as VehicleState;
    if (vehicleState.driverType !== undefined && vehicleState.driverType !== null) {
      this.portrait.setRobotType(vehicleState.driverType, state.team);
      this.portrait.setInVehicle(true);
      this.portrait.setVisible(true);
    }
  } else if (state.objectType === ObjectType.OBJECT_CANNON) {
    this.portrait.setRobotType(RobotType.GRUNT, state.team);
    this.portrait.setInVehicle(true);
    this.portrait.setVisible(true);
