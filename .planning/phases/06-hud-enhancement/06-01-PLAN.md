5. Implement animation playback:
```typescript
public startAnim(animType: PortraitAnimType): void {
  const anim = PORTRAIT_ANIMATIONS.get(animType);
  if (!anim || anim.frames.length === 0) return;

  this.currentAnim = animType;
  this.animStartTime = this.scene.time.now;
  this.currentFrame = anim.frames[0];
  this.applyFrame(this.currentFrame);
}

public update(time: number, delta: number): void {
  if (this.currentAnim !== null) {
    this.processAnimation(time);
  } else if (this.doRandomAnims && time >= this.nextRandomAnimTime) {
    this.startRandomAnim();
  }
}

private processAnimation(time: number): void {
  const anim = PORTRAIT_ANIMATIONS.get(this.currentAnim!);
  if (!anim) return;

  const elapsed = (time - this.animStartTime) / 1000;  // Convert to seconds

  if (elapsed >= anim.totalDuration) {
    // Animation complete
    this.currentAnim = null;
    this.currentFrame = this.getStillFrame();
    this.applyFrame(this.currentFrame);
    this.nextRandomAnimTime = time + 500 + Math.random() * 5000;
    return;
  }

  // Find current frame
  let accumulated = 0;
  for (const frame of anim.frames) {
    if (accumulated + frame.duration > elapsed) {
      if (this.currentFrame !== frame) {
        this.currentFrame = frame;
        this.applyFrame(frame);
      }
      break;
    }
    accumulated += frame.duration;
  }
}

private applyFrame(frame: PortraitFrame): void {
  const faceId = ROBOT_FACE_ID[this.robotType];
  const teamStr = TeamType[this.team].toLowerCase();
  const robotStr = RobotType[this.robotType].toLowerCase();
  const prefix = `portraits/${robotStr}_${teamStr}/SHEADBI${faceId}`;

  // Update head position and texture
  this.head?.setPosition(4, frame.headY);
  const headIndex = 1 + frame.lookDirection;  // 1=straight, 2=right, 3=left
  this.head?.setTexture('hud', `${prefix}_${headIndex.toString().padStart(4, '0')}`);

  // Only render eyes/mouth for LOOK_STRAIGHT
  if (frame.lookDirection === LookDirection.STRAIGHT) {
    const faceOffsetY = FACE_OFFSET_Y[this.robotType];

    // Eyes position: 14 + headX, 8 + headY + faceOffset
    const eyesIndex = 20 + frame.eyes;
    this.eyes?.setTexture('hud', `${prefix}_${eyesIndex.toString().padStart(4, '0')}`);
    this.eyes?.setPosition(14 + 4, 8 + frame.headY + faceOffsetY);
    this.eyes?.setVisible(true);

    // Mouth position: 22 + headX, 24 + headY + faceOffset
    const mouthIndex = 4 + frame.mouth;
    this.mouth?.setTexture('hud', `${prefix}_${mouthIndex.toString().padStart(4, '0')}`);
    this.mouth?.setPosition(22 + 4, 24 + frame.headY + faceOffsetY);
    this.mouth?.setVisible(true);
  } else {
    this.eyes?.setVisible(false);
    this.mouth?.setVisible(false);
  }

  // Hand
  if (frame.handDoRender) {
    const handIndex = 31 + frame.hand;
    this.hand?.setTexture('hud', `${prefix}_${handIndex.toString().padStart(4, '0')}`);
    this.hand?.setPosition(frame.handX - 4, frame.handY - 4);  // Offset from C source
    this.hand?.setVisible(true);
  } else {
    this.hand?.setVisible(false);
  }
}
```
