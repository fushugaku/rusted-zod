4. Implement health bar rendering (matching zhud.cpp RenderHealth):
```typescript
private drawHealthBar(current: number, max: number, maxPossible: number = 100): void {
  this.healthBarBg.clear();
  this.healthBarFill.clear();

  // Calculate widths (matching C source max_dist = 74)
  const healthPercent = Math.max(0, Math.min(1, current / maxPossible));
  const maxPercent = Math.max(0, Math.min(1, max / maxPossible));

  const greenWidth = Math.max(1, HEALTH_BAR_WIDTH * healthPercent);
  const yellowWidth = Math.max(0, (HEALTH_BAR_WIDTH * maxPercent) - greenWidth);
  const grayWidth = HEALTH_BAR_WIDTH - greenWidth - yellowWidth;

  // Draw segments (order: green, yellow, gray)
  let xPos = 0;

  // Green (current health)
  if (greenWidth > 0) {
    this.healthBarFill.fillStyle(COLORS.healthFull, 1);
    this.healthBarFill.fillRect(xPos, 0, greenWidth, HEALTH_BAR_HEIGHT);
    xPos += greenWidth;
  }

  // Yellow (damaged but repairable)
  if (yellowWidth > 0) {
    this.healthBarFill.fillStyle(COLORS.healthLost, 1);
    this.healthBarFill.fillRect(xPos, 0, yellowWidth, HEALTH_BAR_HEIGHT);
    xPos += yellowWidth;
  }

  // Gray (max health lost)
  if (grayWidth > 0) {
    this.healthBarBg.fillStyle(COLORS.healthEmpty, 1);
    this.healthBarBg.fillRect(xPos, 0, grayWidth, HEALTH_BAR_HEIGHT);
  }

  // Update text
  this.healthText.setText(`${Math.round(current)}/${Math.round(max)}`);
}

private drawDriverHealthBar(current: number, max: number): void {
  this.driverHealthBar.clear();

  const percent = Math.max(0, Math.min(1, current / max));
  const width = 30 * percent;

  // Background
  this.driverHealthBar.fillStyle(COLORS.healthEmpty, 1);
  this.driverHealthBar.fillRect(0, 2, 30, 6);

  // Fill
  if (width > 0) {
    this.driverHealthBar.fillStyle(COLORS.driverHealth, 1);
    this.driverHealthBar.fillRect(0, 2, width, 6);
  }

  this.driverHealthText.setText(`${Math.round(current)}`);
}
```

5. Implement state update method:
```typescript
public updateFromState(state: GameObjectState | null): void {
  if (!state) {
    this.setVisible(false);
    return;
  }

  this.setVisible(true);

  // Update based on object type
  switch (state.objectType) {
    case ObjectType.OBJECT_ROBOT:
      this.updateRobotInfo(state as RobotState);
      break;
    case ObjectType.OBJECT_VEHICLE:
      this.updateVehicleInfo(state as VehicleState);
      break;
    case ObjectType.OBJECT_CANNON:
      this.updateCannonInfo(state as CannonState);
      break;
    case ObjectType.OBJECT_BUILDING:
      this.updateBuildingInfo(state as BuildingObjectState);
      break;
    default:
      this.setVisible(false);
  }
}

private updateRobotInfo(state: RobotState): void {
  this.unitTypeLabel.setText("ROBOT");
  this.unitNameLabel.setText(RobotType[state.robotType]?.toUpperCase() ?? "UNKNOWN");

  // Health bar
  this.drawHealthBar(state.health, state.maxHealth);

  // Grenades
  this.grenadeContainer.setVisible(true);
  this.grenadeCountText.setText(state.grenades.toString().padStart(2, '0'));
  // Update grenade icon texture based on team
  const teamStr = TeamType[state.team].toLowerCase();
  this.grenadeIcon.setTexture('hud', `icon_grenade_${teamStr}`);

  // Hide driver health for robots
  this.driverHealthContainer.setVisible(false);

  // Weapon info
  this.weaponText.setText(this.getRobotWeapon(state.robotType));
}

private updateVehicleInfo(state: VehicleState): void {
  this.unitTypeLabel.setText("VEHICLE");
  this.unitNameLabel.setText(VehicleType[state.vehicleType]?.replace(/_/g, " ")?.toUpperCase() ?? "UNKNOWN");

  // Vehicle health
  this.drawHealthBar(state.health, state.maxHealth);

  // Driver health (if has driver)
  if (state.hasDriver && state.driverHealth !== undefined && state.driverMaxHealth !== undefined) {
    this.driverHealthContainer.setVisible(true);
    this.drawDriverHealthBar(state.driverHealth, state.driverMaxHealth);
  } else {
    this.driverHealthContainer.setVisible(false);
  }

  // No grenades for vehicles
  this.grenadeContainer.setVisible(false);

  // Weapon info
  this.weaponText.setText(this.getVehicleWeapon(state.vehicleType));
}

private updateCannonInfo(state: CannonState): void {
  this.unitTypeLabel.setText("CANNON");
  this.unitNameLabel.setText(CannonType[state.cannonType]?.replace(/_/g, " ")?.toUpperCase() ?? "UNKNOWN");

  this.drawHealthBar(state.health, state.maxHealth);
  this.driverHealthContainer.setVisible(false);
  this.grenadeContainer.setVisible(false);

  this.weaponText.setText(this.getCannonWeapon(state.cannonType));
}

private updateBuildingInfo(state: BuildingObjectState): void {
  this.unitTypeLabel.setText("BUILDING");
  this.unitNameLabel.setText(BuildingType[state.buildingType]?.replace(/_/g, " ")?.toUpperCase() ?? "UNKNOWN");

  this.drawHealthBar(state.health, state.maxHealth);
  this.driverHealthContainer.setVisible(false);
  this.grenadeContainer.setVisible(false);

  this.weaponText.setText(`Level ${state.level}`);
}
```
