```typescript
/**
 * RallyPointVisualizer - Shows rally points for production buildings
 * Displays flag marker at rally location with line from building
 */

import Phaser from "phaser";
import { TeamType } from "@/types";

// ============================================================================
// CONSTANTS
// ============================================================================

const FLAG_SIZE = 12;
const LINE_WIDTH = 2;
const FLAG_POLE_HEIGHT = 16;

// ============================================================================
// RALLY POINT VISUALIZER CLASS
// ============================================================================

export class RallyPointVisualizer {
  private scene: Phaser.Scene;
  private container: Phaser.GameObjects.Container;
  private flagGraphics: Phaser.GameObjects.Graphics;
  private lineGraphics: Phaser.GameObjects.Graphics;

  private buildingX: number = 0;
  private buildingY: number = 0;
  private rallyX: number = 0;
  private rallyY: number = 0;
  private team: TeamType = TeamType.NULL;
  private isVisible: boolean = false;

  constructor(scene: Phaser.Scene) {
    this.scene = scene;

    // Create container for rally point elements
    this.container = scene.add.container(0, 0);
    this.container.setDepth(50); // Above terrain, below units

    // Create graphics for line
    this.lineGraphics = scene.add.graphics();
    this.container.add(this.lineGraphics);

    // Create graphics for flag
    this.flagGraphics = scene.add.graphics();
    this.container.add(this.flagGraphics);

    this.hide();
  }

  /**
   * Show rally point for a building
   */
  public showRallyPoint(
    buildingX: number,
    buildingY: number,
    rallyX: number,
    rallyY: number,
    team: TeamType
  ): void {
    this.buildingX = buildingX;
    this.buildingY = buildingY;
    this.rallyX = rallyX;
    this.rallyY = rallyY;
    this.team = team;
    this.isVisible = true;

    this.render();
    this.container.setVisible(true);
  }

  /**
   * Update rally point position (for dragging or new position)
   */
  public updateRallyPosition(rallyX: number, rallyY: number): void {
    this.rallyX = rallyX;
    this.rallyY = rallyY;
    this.render();
  }

  /**
   * Hide rally point visualization
   */
  public hide(): void {
    this.isVisible = false;
    this.container.setVisible(false);
  }

  /**
   * Render rally point elements
   */
  private render(): void {
    const teamColor = this.getTeamColor();

    // Clear previous graphics
    this.lineGraphics.clear();
    this.flagGraphics.clear();

    // Draw dashed line from building to rally point
    this.lineGraphics.lineStyle(LINE_WIDTH, teamColor, 0.6);
    this.drawDashedLine(
      this.buildingX,
      this.buildingY,
      this.rallyX,
      this.rallyY,
      8, // dash length
      4  // gap length
    );

    // Draw flag at rally point
    this.drawFlag(this.rallyX, this.rallyY, teamColor);
  }

  /**
   * Draw a dashed line
   */
  private drawDashedLine(
    x1: number,
    y1: number,
    x2: number,
    y2: number,
    dashLength: number,
    gapLength: number
  ): void {
    const dx = x2 - x1;
    const dy = y2 - y1;
    const distance = Math.sqrt(dx * dx + dy * dy);
    const dashCount = Math.floor(distance / (dashLength + gapLength));

    const unitX = dx / distance;
    const unitY = dy / distance;

    for (let i = 0; i < dashCount; i++) {
      const startX = x1 + unitX * i * (dashLength + gapLength);
      const startY = y1 + unitY * i * (dashLength + gapLength);
      const endX = startX + unitX * dashLength;
      const endY = startY + unitY * dashLength;

      this.lineGraphics.beginPath();
      this.lineGraphics.moveTo(startX, startY);
      this.lineGraphics.lineTo(endX, endY);
      this.lineGraphics.strokePath();
    }
  }

  /**
   * Draw flag marker
   */
  private drawFlag(x: number, y: number, color: number): void {
    // Flag pole
    this.flagGraphics.lineStyle(2, 0x444444, 1);
    this.flagGraphics.beginPath();
    this.flagGraphics.moveTo(x, y);
    this.flagGraphics.lineTo(x, y - FLAG_POLE_HEIGHT);
    this.flagGraphics.strokePath();

    // Flag triangle
    this.flagGraphics.fillStyle(color, 0.9);
    this.flagGraphics.beginPath();
    this.flagGraphics.moveTo(x, y - FLAG_POLE_HEIGHT);
    this.flagGraphics.lineTo(x + FLAG_SIZE, y - FLAG_POLE_HEIGHT + FLAG_SIZE / 2);
    this.flagGraphics.lineTo(x, y - FLAG_POLE_HEIGHT + FLAG_SIZE);
    this.flagGraphics.closePath();
    this.flagGraphics.fillPath();

    // Flag border
    this.flagGraphics.lineStyle(1, 0x000000, 0.5);
    this.flagGraphics.strokePath();

    // Base circle
    this.flagGraphics.fillStyle(0x333333, 1);
    this.flagGraphics.fillCircle(x, y, 3);
  }

  /**
   * Get team color
   */
  private getTeamColor(): number {
    switch (this.team) {
      case TeamType.RED: return 0xff4444;
      case TeamType.BLUE: return 0x4444ff;
      case TeamType.GREEN: return 0x44ff44;
      case TeamType.YELLOW: return 0xffff44;
      case TeamType.PURPLE: return 0x884488;
      case TeamType.TEAL: return 0x448888;
      case TeamType.WHITE: return 0xeeeeee;
      case TeamType.BLACK: return 0x444444;
      default: return 0x888888;
    }
  }

  /**
   * Check if point is near rally flag (for interaction)
   */
  public isNearRallyPoint(x: number, y: number, threshold: number = 16): boolean {
    if (!this.isVisible) return false;
    const dx = x - this.rallyX;
    const dy = y - this.rallyY;
    return Math.sqrt(dx * dx + dy * dy) < threshold;
  }

  /**
   * Get current rally position
   */
  public getRallyPosition(): { x: number; y: number } {
    return { x: this.rallyX, y: this.rallyY };
  }

  /**
   * Check if currently visible
   */
  public isShowing(): boolean {
    return this.isVisible;
  }

  /**
   * Destroy the visualizer
   */
  public destroy(): void {
    this.container.destroy();
  }
}

export default RallyPointVisualizer;
```
