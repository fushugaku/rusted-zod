/**
 * CraneConstructionEffect - Animated construction scene for crane repairs
 * Ported from source/ecraneconco.cpp
 *
 * Effect elements:
 * - Construction barricade (conco) - 8 animation frames
 * - Traffic cones (cone0, cone1)
 * - Warning sign with flip animation
 * - Worker with jackhammer (2 frames, 45ms timing)
 * - Worker with paper/pointing (2 frames + 3 pointing frames)
 * - Travel animations for workers going to/from site
 */

import Phaser from "phaser";
import { TeamType } from "@/types";

// ============================================================================
// CONSTANTS (from ecraneconco.cpp)
// ============================================================================

const TRAVEL_TIME = 800; // ms (0.8 seconds in original)
const JACKHAMMER_FRAME_TIME = 45; // ms
const PAPER_FRAME_TIME = 150; // ms
const PAPER_POINTING_TIME = 300; // ms

// Positioning constants
const CONCO_DIST_FROM_ENTRANCE = 12;
const CONE_DIST_FROM_ENTRANCE = 6;
const CONE_DIST_FROM_CENTER = 18;
const SIGN_DIST_FROM_CONCO = 6;

// ============================================================================
// INTERFACES
// ============================================================================

interface RenderItem {
  type: string;
  x: number;
  y: number;
  startX: number;
  startY: number;
  destX: number;
  destY: number;
  width: number;
  height: number;
}

// ============================================================================
// CRANE CONSTRUCTION EFFECT CLASS
// ============================================================================

export class CraneConstructionEffect {
  private scene: Phaser.Scene;
  private container: Phaser.GameObjects.Container;
  private team: TeamType;

  // State
  private isActive: boolean = false;
  private travelingTo: boolean = false;
  private travelingBack: boolean = false;
  private travelStartTime: number = 0;
  private travelProgress: number = 0;

  // Animation state
  private concoFrame: number = 7;
  private signFrame: number = 7;
  private jackhammerFrame: number = 0;
  private paperFrame: number = 0;
  private isPaperPointing: boolean = false;
  private nextJackhammerTime: number = 0;
  private nextPaperTime: number = 0;

  // Render items
  private renderItems: Map<string, RenderItem> = new Map();

  // Graphics objects
  private concoGraphics: Phaser.GameObjects.Graphics;
  private cone0Graphics: Phaser.GameObjects.Graphics;
  private cone1Graphics: Phaser.GameObjects.Graphics;
  private signGraphics: Phaser.GameObjects.Graphics;
  private jackbotGraphics: Phaser.GameObjects.Graphics;
  private paperbotGraphics: Phaser.GameObjects.Graphics;

  // Positions
  private craneX: number = 0;
  private craneY: number = 0;
  private buildingX: number = 0;
  private buildingY: number = 0;
  private buildingW: number = 0;
  private buildingH: number = 0;

  constructor(scene: Phaser.Scene) {
    this.scene = scene;
    this.team = TeamType.NULL;

    // Create container
    this.container = scene.add.container(0, 0);
    this.container.setDepth(75); // Above terrain, below most units

    // Create graphics objects for each element
    this.concoGraphics = scene.add.graphics();
    this.cone0Graphics = scene.add.graphics();
    this.cone1Graphics = scene.add.graphics();
    this.signGraphics = scene.add.graphics();
    this.jackbotGraphics = scene.add.graphics();
    this.paperbotGraphics = scene.add.graphics();

    this.container.add([
      this.concoGraphics,
      this.cone0Graphics,
      this.cone1Graphics,
      this.signGraphics,
      this.jackbotGraphics,
      this.paperbotGraphics,
    ]);

    this.container.setVisible(false);
  }

  /**
   * Start construction effect
   * @param craneX Crane position X
   * @param craneY Crane position Y
   * @param buildingX Building position X
   * @param buildingY Building position Y
   * @param buildingW Building width
   * @param buildingH Building height
   * @param team Crane's team
   */
  public start(
    craneX: number,
    craneY: number,
    buildingX: number,
    buildingY: number,
    buildingW: number,
    buildingH: number,
    team: TeamType
  ): void {
    this.craneX = craneX;
    this.craneY = craneY;
    this.buildingX = buildingX;
    this.buildingY = buildingY;
    this.buildingW = buildingW;
    this.buildingH = buildingH;
    this.team = team;

    // Calculate center points
    const craneCX = craneX + 16;
    const craneCY = craneY + 16;
    const buildingCX = buildingX + buildingW / 2;
    const buildingCY = buildingY + buildingH / 2;

    // Initialize render items with start at crane and destination at building
    this.initializeRenderItems(craneCX, craneCY, buildingCX, buildingY, buildingH);

    // Start travel animation
    this.isActive = true;
    this.travelingTo = true;
    this.travelingBack = false;
    this.travelStartTime = this.scene.time.now;
    this.concoFrame = 7;

    this.container.setVisible(true);
  }

  /**
   * Initialize render item positions
   */
  private initializeRenderItems(
    craneCX: number,
    craneCY: number,
    buildingCX: number,
    buildingY: number,
    buildingH: number
  ): void {
    // Conco (barricade) - positioned below building entrance
    this.renderItems.set("conco", {
      type: "conco",
      x: craneCX,
      y: craneCY,
      startX: craneCX,
      startY: craneCY,
      destX: buildingCX - 24, // ~half conco width
      destY: buildingY + buildingH + CONCO_DIST_FROM_ENTRANCE,
      width: 48,
      height: 32,
    });

    // Cones
    this.renderItems.set("cone0", {
      type: "cone",
      x: craneCX,
      y: craneCY,
      startX: craneCX,
      startY: craneCY,
      destX: buildingCX - 8 - CONE_DIST_FROM_CENTER,
      destY: buildingY + buildingH + CONE_DIST_FROM_ENTRANCE,
      width: 8,
      height: 12,
    });

    this.renderItems.set("cone1", {
      type: "cone",
      x: craneCX,
      y: craneCY,
      startX: craneCX,
      startY: craneCY,
      destX: buildingCX + CONE_DIST_FROM_CENTER,
      destY: buildingY + buildingH + CONE_DIST_FROM_ENTRANCE,
      width: 8,
      height: 12,
    });

    // Sign
    const concoItem = this.renderItems.get("conco")!;
    this.renderItems.set("sign", {
      type: "sign",
      x: craneCX,
      y: craneCY,
      startX: craneCX,
      startY: craneCY,
      destX: concoItem.destX + 24 - 8, // Center above conco
      destY: concoItem.destY - 16 - 1,
      width: 16,
      height: 16,
    });

    // Workers - random positions around the work area
    this.renderItems.set("jackbot", {
      type: "jackbot",
      x: craneCX,
      y: craneCY,
      startX: craneCX,
      startY: craneCY,
      destX: buildingCX - 16 + Math.random() * 32,
      destY: buildingY + buildingH + 16 + Math.random() * 32,
      width: 16,
      height: 16,
    });

    this.renderItems.set("paperbot", {
      type: "paperbot",
      x: craneCX,
      y: craneCY,
      startX: craneCX,
      startY: craneCY,
      destX: buildingCX - 16 + Math.random() * 32,
      destY: buildingY + buildingH + 16 + Math.random() * 32,
      width: 16,
      height: 16,
    });
  }

  /**
   * Begin return animation (construction complete)
   */
  public beginReturn(): void {
    if (!this.isActive) return;

    // Set up return trip
    this.travelingTo = false;
    this.travelingBack = true;
    this.travelStartTime = this.scene.time.now;

    // Swap start and dest for each item
    this.renderItems.forEach((item) => {
      const tempX = item.startX;
      const tempY = item.startY;
      item.startX = item.x;
      item.startY = item.y;
      item.destX = tempX;
      item.destY = tempY;
    });
  }

  /**
   * Update effect
   */
  public update(time: number, _delta: number): void {
    if (!this.isActive) return;

    const elapsed = time - this.travelStartTime;

    // Update travel animation
    if (this.travelingTo || this.travelingBack) {
      this.travelProgress = Math.min(1, elapsed / TRAVEL_TIME);

      // Update frame based on progress
      if (this.travelingTo) {
        this.concoFrame = Math.floor(7 * (1 - this.travelProgress));
      } else {
        this.concoFrame = Math.floor(7 * this.travelProgress);
      }
      this.signFrame = this.concoFrame;

      // Move all items
      this.renderItems.forEach((item) => {
        item.x = item.startX + (item.destX - item.startX) * this.travelProgress;
        item.y = item.startY + (item.destY - item.startY) * this.travelProgress;
      });

      // Check if travel complete
      if (this.travelProgress >= 1) {
        if (this.travelingTo) {
          this.travelingTo = false;
          this.concoFrame = 0;
        } else if (this.travelingBack) {
          // Effect complete
          this.stop();
          return;
        }
      }
    }

    // Update worker animations (only when not traveling)
    if (!this.travelingTo && !this.travelingBack) {
      // Jackhammer animation
      if (time >= this.nextJackhammerTime) {
        this.nextJackhammerTime = time + JACKHAMMER_FRAME_TIME + Math.random() * 20;
        this.jackhammerFrame = this.jackhammerFrame === 0 ? 1 : 0;
      }

      // Paper/pointing animation
      if (time >= this.nextPaperTime) {
        if (this.isPaperPointing) {
          this.paperFrame++;
          if (this.paperFrame >= 3) {
            this.paperFrame = 0;
            this.isPaperPointing = false;
          }
          this.nextPaperTime = time + PAPER_POINTING_TIME + Math.random() * 100;
        } else {
          // Random chance to start pointing
          if (Math.random() < 0.1) {
            this.paperFrame = 0;
            this.isPaperPointing = true;
          } else {
            this.paperFrame = this.paperFrame === 0 ? 1 : 0;
          }
          this.nextPaperTime = time + PAPER_FRAME_TIME + Math.random() * 100;
        }
      }
    }

    // Render
    this.render();
  }

  /**
   * Render all effect elements
   */
  private render(): void {
    const teamColor = this.getTeamColor();

    // Clear all graphics
    this.concoGraphics.clear();
    this.cone0Graphics.clear();
    this.cone1Graphics.clear();
    this.signGraphics.clear();
    this.jackbotGraphics.clear();
    this.paperbotGraphics.clear();

    // Sort items by Y for proper z-ordering
    const sortedItems = Array.from(this.renderItems.values()).sort(
      (a, b) => a.y + a.height - (b.y + b.height)
    );

    for (const item of sortedItems) {
      switch (item.type) {
        case "conco":
          this.renderConco(item, teamColor);
          break;
        case "cone":
          this.renderCone(item, teamColor);
          break;
        case "sign":
          this.renderSign(item, teamColor);
          break;
        case "jackbot":
          this.renderJackbot(item, teamColor);
          break;
        case "paperbot":
          this.renderPaperbot(item, teamColor);
          break;
      }
    }
  }

  /**
   * Render construction barricade
   */
  private renderConco(item: RenderItem, teamColor: number): void {
    // Scale based on frame (0 = full, 7 = collapsed)
    const scale = 1 - this.concoFrame * 0.1;

    this.concoGraphics.fillStyle(teamColor, 0.8);
    this.concoGraphics.fillRect(
      item.x - (item.width * scale) / 2,
      item.y,
      item.width * scale,
      item.height * scale
    );

    // Stripes
    this.concoGraphics.fillStyle(0xffff00, 0.9);
    for (let i = 0; i < 3; i++) {
      this.concoGraphics.fillRect(
        item.x - (item.width * scale) / 2 + i * 16,
        item.y + 4,
        8 * scale,
        item.height * scale - 8
      );
    }
  }

  /**
   * Render traffic cone
   */
  private renderCone(item: RenderItem, teamColor: number): void {
    // Draw cone (triangle)
    this.cone0Graphics.fillStyle(0xff6600, 1);
    this.cone0Graphics.beginPath();
    this.cone0Graphics.moveTo(item.x, item.y + item.height);
    this.cone0Graphics.lineTo(item.x + item.width / 2, item.y);
    this.cone0Graphics.lineTo(item.x + item.width, item.y + item.height);
    this.cone0Graphics.closePath();
    this.cone0Graphics.fillPath();

    // White stripe
    this.cone0Graphics.fillStyle(0xffffff, 0.9);
    this.cone0Graphics.fillRect(item.x + 1, item.y + item.height / 2, item.width - 2, 3);
  }

  /**
   * Render warning sign
   */
  private renderSign(item: RenderItem, teamColor: number): void {
    const scale = 1 - this.signFrame * 0.1;

    // Sign background
    this.signGraphics.fillStyle(teamColor, 0.9);
    this.signGraphics.fillRect(
      item.x,
      item.y,
      item.width * scale,
      item.height * scale
    );

    // Sign post
    this.signGraphics.fillStyle(0x444444, 1);
    this.signGraphics.fillRect(
      item.x + item.width / 2 - 1,
      item.y + item.height * scale,
      2,
      8
    );
  }

  /**
   * Render jackhammer worker
   */
  private renderJackbot(item: RenderItem, teamColor: number): void {
    const isMoving = this.travelingTo || this.travelingBack;

    // Worker body (circle)
    this.jackbotGraphics.fillStyle(teamColor, 1);
    this.jackbotGraphics.fillCircle(item.x + 8, item.y + 8, 6);

    // Hard hat
    this.jackbotGraphics.fillStyle(0xffff00, 1);
    this.jackbotGraphics.fillCircle(item.x + 8, item.y + 4, 4);

    if (!isMoving) {
      // Jackhammer
      const jackY = this.jackhammerFrame === 0 ? item.y + 12 : item.y + 14;
      this.jackbotGraphics.fillStyle(0x666666, 1);
      this.jackbotGraphics.fillRect(item.x + 6, jackY, 4, 8);
    }
  }

  /**
   * Render paper/pointing worker
   */
  private renderPaperbot(item: RenderItem, teamColor: number): void {
    const isMoving = this.travelingTo || this.travelingBack;

    // Worker body
    this.paperbotGraphics.fillStyle(teamColor, 1);
    this.paperbotGraphics.fillCircle(item.x + 8, item.y + 8, 6);

    // Hard hat
    this.paperbotGraphics.fillStyle(0xffffff, 1);
    this.paperbotGraphics.fillCircle(item.x + 8, item.y + 4, 4);

    if (!isMoving) {
      if (this.isPaperPointing) {
        // Pointing arm
        this.paperbotGraphics.lineStyle(2, teamColor, 1);
        this.paperbotGraphics.beginPath();
        this.paperbotGraphics.moveTo(item.x + 10, item.y + 8);
        this.paperbotGraphics.lineTo(item.x + 16 + this.paperFrame * 2, item.y + 6);
        this.paperbotGraphics.strokePath();
      } else {
        // Clipboard
        this.paperbotGraphics.fillStyle(0xeeeeee, 1);
        this.paperbotGraphics.fillRect(item.x + 2, item.y + 10, 6, 8);
        this.paperbotGraphics.fillStyle(0x666666, 1);
        this.paperbotGraphics.fillRect(item.x + 3, item.y + 12 + this.paperFrame, 4, 1);
      }
    }
  }

  /**
   * Get team color
   */
  private getTeamColor(): number {
    switch (this.team) {
      case TeamType.RED: return 0xff4444;
      case TeamType.BLUE: return 0x4444ff;
      case TeamType.GREEN: return 0x44ff44;
      case TeamType.YELLOW: return 0xffff44;
      case TeamType.PURPLE: return 0x884488;
      case TeamType.TEAL: return 0x448888;
      case TeamType.WHITE: return 0xeeeeee;
      case TeamType.BLACK: return 0x444444;
      default: return 0x888888;
    }
  }

  /**
   * Stop and hide effect
   */
  public stop(): void {
    this.isActive = false;
    this.travelingTo = false;
    this.travelingBack = false;
    this.container.setVisible(false);
  }

  /**
   * Check if effect is active
   */
  public isRunning(): boolean {
    return this.isActive;
  }

  /**
   * Destroy the effect
   */
  public destroy(): void {
    this.container.destroy();
  }
}

export default CraneConstructionEffect;
