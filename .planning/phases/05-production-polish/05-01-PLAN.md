```typescript
/**
 * Get detailed modifier information for a building
 * Used by UI to display zone bonus and damage penalty
 */
public getModifierInfo(buildingRefId: number): ProductionModifiers | null {
  const building = this.buildings.get(buildingRefId);
  if (!building) return null;

  // Get base build time for current or next unit
  let baseBuildTime = 0;
  if (building.currentUnit) {
    baseBuildTime = this.getBaseBuildTime(
      building.buildingType,
      building.level,
      building.currentUnit.objectType,
      building.currentUnit.objectId
    );
  } else if (building.queue.length > 0) {
    const nextUnit = building.queue[0];
    baseBuildTime = this.getBaseBuildTime(
      building.buildingType,
      building.level,
      nextUnit.objectType,
      nextUnit.objectId
    );
  }

  // Calculate zone bonus (reduces time)
  const zoneBonus = 0.5 * building.zoneOwnership;

  // Calculate damage penalty (increases time)
  const healthPercent = building.health / building.maxHealth;
  const damageBonus = 1.25 * (1.0 - healthPercent);

  // Calculate effective build time
  let effectiveBuildTime = baseBuildTime;
  effectiveBuildTime -= baseBuildTime * zoneBonus;
  effectiveBuildTime += effectiveBuildTime * damageBonus;

  return {
    zoneBonus,
    damageBonus,
    totalModifier: 1 - zoneBonus + damageBonus,
    effectiveBuildTime,
    baseBuildTime,
  };
}
```

4. Add updateModifiers method:
```typescript
/**
 * Update modifier display
 */
public updateModifiers(modifiers: ProductionModifiers | null): void {
  if (!modifiers) {
    this.zoneBonusText.setText("");
    this.damagePenaltyText.setText("");
    this.estimatedTimeText.setText("");
    return;
  }

  // Zone bonus display
  if (modifiers.zoneBonus > 0) {
    const bonusPercent = Math.round(modifiers.zoneBonus * 100);
    this.zoneBonusText.setText(`+${bonusPercent}% Zone Bonus`);
    this.zoneBonusText.setVisible(true);
  } else {
    this.zoneBonusText.setVisible(false);
  }

  // Damage penalty display
  if (modifiers.damageBonus > 0.01) {
    const penaltyPercent = Math.round(modifiers.damageBonus * 100);
    this.damagePenaltyText.setText(`-${penaltyPercent}% Damage Penalty`);
    this.damagePenaltyText.setVisible(true);
  } else {
    this.damagePenaltyText.setVisible(false);
  }

  // Estimated time display (formatted as M:SS)
  if (modifiers.effectiveBuildTime > 0) {
    const totalSeconds = Math.ceil(modifiers.effectiveBuildTime / 1000);
    const minutes = Math.floor(totalSeconds / 60);
    const seconds = totalSeconds % 60;
    this.estimatedTimeText.setText(`Est: ${minutes}:${seconds.toString().padStart(2, '0')}`);
  } else {
    this.estimatedTimeText.setText("");
  }
}
```
