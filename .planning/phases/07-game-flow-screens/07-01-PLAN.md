```typescript
import Phaser from "phaser";
import { TeamType } from "@/types";
import { TEAM_COLORS } from "@/config/GameConfig";
import type { GameResult, TeamStats } from "@/game/GameStateSystem";

export class VictoryScene extends Phaser.Scene {
  private gameResult: GameResult | null = null;

  constructor() {
    super({ key: "VictoryScene" });
  }

  init(data: { result: GameResult }): void {
    this.gameResult = data.result;
  }

  create(): void {
    const { width, height } = this.scale;

    // Dark overlay background
    this.add.rectangle(width / 2, height / 2, width, height, 0x000000, 0.85);

    // Victory header with team color
    const teamColor = this.gameResult?.winner !== null
      ? TEAM_COLORS[this.gameResult.winner] ?? 0x00ff00
      : 0x00ff00;

    this.add.text(width / 2, 80, "VICTORY!", {
      fontFamily: "Courier New",
      fontSize: "64px",
      color: `#${teamColor.toString(16).padStart(6, "0")}`,
      fontStyle: "bold",
    }).setOrigin(0.5);

    // Victory condition subtitle
    const conditionText = this.getConditionText();
    this.add.text(width / 2, 140, conditionText, {
      fontFamily: "Courier New",
      fontSize: "20px",
      color: "#aaaaaa",
    }).setOrigin(0.5);

    // Stats panel
    this.createStatsPanel(width / 2, height / 2);

    // Return to menu button
    this.createMenuButton(width / 2, height - 80);

    // Play victory sound if available
    // this.sound.play('victory'); // Add when sound asset exists
  }
```

```typescript
private createStatsPanel(centerX: number, centerY: number): void {
  if (!this.gameResult) return;

  const panelWidth = 400;
  const panelHeight = 250;
  const panelX = centerX - panelWidth / 2;
  const panelY = centerY - panelHeight / 2;

  // Panel background
  const panel = this.add.graphics();
  panel.fillStyle(0x1a1a2e, 0.95);
  panel.fillRoundedRect(panelX, panelY, panelWidth, panelHeight, 8);
  panel.lineStyle(2, 0x3a5a3a, 1);
  panel.strokeRoundedRect(panelX, panelY, panelWidth, panelHeight, 8);

  // Stats header
  this.add.text(centerX, panelY + 20, "GAME STATISTICS", {
    fontFamily: "Courier New",
    fontSize: "18px",
    color: "#00ff66",
    fontStyle: "bold",
  }).setOrigin(0.5);

  // Get player stats
  const playerStats = this.gameResult.teamStats.get(this.gameResult.winner ?? TeamType.RED);

  // Stats list
  const stats = [
    { label: "Game Duration", value: this.formatDuration(this.gameResult.gameDuration) },
    { label: "Units Killed", value: playerStats?.unitsKilled.toString() ?? "0" },
    { label: "Units Lost", value: playerStats?.unitsLost.toString() ?? "0" },
    { label: "Zones Controlled", value: playerStats?.zonesOwned.toString() ?? "0" },
    { label: "Buildings Owned", value: playerStats?.buildingsOwned.toString() ?? "0" },
  ];

  let yOffset = panelY + 60;
  const labelX = panelX + 40;
  const valueX = panelX + panelWidth - 40;

  for (const stat of stats) {
    this.add.text(labelX, yOffset, stat.label, {
      fontFamily: "Courier New",
      fontSize: "16px",
      color: "#cccccc",
    }).setOrigin(0, 0.5);

    this.add.text(valueX, yOffset, stat.value, {
      fontFamily: "Courier New",
      fontSize: "16px",
      color: "#ffffff",
      fontStyle: "bold",
    }).setOrigin(1, 0.5);

    yOffset += 35;
  }
}
```

```typescript
private formatDuration(seconds: number): string {
  const mins = Math.floor(seconds / 60);
  const secs = Math.floor(seconds % 60);
  return `${mins}:${secs.toString().padStart(2, "0")}`;
}

private getConditionText(): string {
  if (!this.gameResult) return "";

  switch (this.gameResult.condition) {
    case 0: return "Total Domination - All zones captured!";
    case 1: return "Zone Majority - Held territory long enough!";
    case 2: return "Elimination - All enemies destroyed!";
    case 3: return "Time Limit - Most territory at end!";
    default: return "Victory achieved!";
  }
}

private createMenuButton(x: number, y: number): void {
  const buttonWidth = 200;
  const buttonHeight = 50;

  const button = this.add.graphics();
  button.fillStyle(0x3a5a3a, 1);
  button.fillRoundedRect(-buttonWidth / 2, -buttonHeight / 2, buttonWidth, buttonHeight, 8);
  button.lineStyle(2, 0x5a8a5a, 1);
  button.strokeRoundedRect(-buttonWidth / 2, -buttonHeight / 2, buttonWidth, buttonHeight, 8);
  button.setPosition(x, y);

  const buttonText = this.add.text(x, y, "MAIN MENU", {
    fontFamily: "Courier New",
    fontSize: "18px",
    color: "#ffffff",
    fontStyle: "bold",
  }).setOrigin(0.5);

  // Make interactive
  button.setInteractive(
    new Phaser.Geom.Rectangle(-buttonWidth / 2, -buttonHeight / 2, buttonWidth, buttonHeight),
    Phaser.Geom.Rectangle.Contains
  );

  button.on("pointerover", () => {
    button.clear();
    button.fillStyle(0x4a7a4a, 1);
    button.fillRoundedRect(-buttonWidth / 2, -buttonHeight / 2, buttonWidth, buttonHeight, 8);
    button.lineStyle(2, 0x6aaa6a, 1);
    button.strokeRoundedRect(-buttonWidth / 2, -buttonHeight / 2, buttonWidth, buttonHeight, 8);
  });

  button.on("pointerout", () => {
    button.clear();
    button.fillStyle(0x3a5a3a, 1);
    button.fillRoundedRect(-buttonWidth / 2, -buttonHeight / 2, buttonWidth, buttonHeight, 8);
    button.lineStyle(2, 0x5a8a5a, 1);
    button.strokeRoundedRect(-buttonWidth / 2, -buttonHeight / 2, buttonWidth, buttonHeight, 8);
  });

  button.on("pointerdown", () => {
    // For now, restart GameScene (MainMenu will be added in 07-02)
    this.scene.stop("HUDScene");
    this.scene.start("GameScene");
  });
}
```
