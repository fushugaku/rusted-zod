```typescript
import Phaser from "phaser";

export class MainMenuScene extends Phaser.Scene {
  constructor() {
    super({ key: "MainMenuScene" });
  }

  create(): void {
    const { width, height } = this.scale;

    // Background
    this.add.rectangle(width / 2, height / 2, width, height, 0x0a0a1a, 1);

    // Game title
    this.add.text(width / 2, 120, "Z.O.D. ENGINE", {
      fontFamily: "Courier New",
      fontSize: "56px",
      color: "#00ff66",
      fontStyle: "bold",
    }).setOrigin(0.5);

    // Subtitle
    this.add.text(width / 2, 180, "TypeScript Port", {
      fontFamily: "Courier New",
      fontSize: "20px",
      color: "#666666",
    }).setOrigin(0.5);

    // Menu buttons
    const buttonY = height / 2;
    const buttonSpacing = 70;

    this.createMenuButton(width / 2, buttonY - buttonSpacing, "PLAY GAME", () => {
      this.scene.start("MapSelectionScene");
    });

    this.createMenuButton(width / 2, buttonY, "SETTINGS", () => {
      this.scene.start("SettingsScene");
    });

    this.createMenuButton(width / 2, buttonY + buttonSpacing, "QUICK START", () => {
      // Start directly with default map and team
      this.scene.start("GameScene", {
        mapPath: "maps/p02_bb_orig01.map",
        team: 1, // TeamType.RED
      });
    });

    // Version info
    this.add.text(width - 10, height - 10, "v0.1.0", {
      fontFamily: "Courier New",
      fontSize: "12px",
      color: "#444444",
    }).setOrigin(1, 1);

    // Credits
    this.add.text(10, height - 10, "Based on original Zod Engine", {
      fontFamily: "Courier New",
      fontSize: "12px",
      color: "#444444",
    }).setOrigin(0, 1);
  }

  private createMenuButton(x: number, y: number, text: string, onClick: () => void): void {
    const buttonWidth = 250;
    const buttonHeight = 50;

    // Button background
    const button = this.add.graphics();
    button.fillStyle(0x1a2a1a, 1);
    button.fillRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
    button.lineStyle(2, 0x3a5a3a, 1);
    button.strokeRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);

    // Button text
    const buttonText = this.add.text(x, y, text, {
      fontFamily: "Courier New",
      fontSize: "20px",
      color: "#ffffff",
      fontStyle: "bold",
    }).setOrigin(0.5);

    // Interactive area
    const hitArea = new Phaser.Geom.Rectangle(
      x - buttonWidth / 2,
      y - buttonHeight / 2,
      buttonWidth,
      buttonHeight
    );
    button.setInteractive(hitArea, Phaser.Geom.Rectangle.Contains);

    // Hover effects
    button.on("pointerover", () => {
      button.clear();
      button.fillStyle(0x2a4a2a, 1);
      button.fillRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
      button.lineStyle(2, 0x5a8a5a, 1);
      button.strokeRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
      buttonText.setColor("#00ff66");
    });

    button.on("pointerout", () => {
      button.clear();
      button.fillStyle(0x1a2a1a, 1);
      button.fillRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
      button.lineStyle(2, 0x3a5a3a, 1);
      button.strokeRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
      buttonText.setColor("#ffffff");
    });

    button.on("pointerdown", onClick);
  }
}
```

```typescript
import Phaser from "phaser";
import { TeamType, PlanetType, PLANET_TYPE_STRINGS } from "@/types";
import type { MapInfo } from "@/types";
import { TEAM_COLORS } from "@/config/GameConfig";
import { MapLoader } from "@/map";

// Available maps (hardcoded list - could be made dynamic with server)
const AVAILABLE_MAPS = [
  "maps/p02_bb_orig01.map",
  "maps/p02_bb_orig02.map",
  "maps/p02_bb_orig03.map",
  "maps/p02_bb_orig04.map",
  "maps/p02_bb_orig05.map",
  "maps/p02_bb_orig06.map",
  "maps/p02_bb_orig07.map",
  "maps/p02_bb_orig08.map",
  "maps/p02_bb_orig09.map",
  "maps/p02_bb_orig10.map",
];

export class MapSelectionScene extends Phaser.Scene {
  private mapInfos: MapInfo[] = [];
  private selectedMapIndex: number = 0;
  private selectedTeam: TeamType = TeamType.RED;
  private mapListContainer!: Phaser.GameObjects.Container;
  private mapInfoPanel!: Phaser.GameObjects.Container;
  private teamButtons: Map<TeamType, Phaser.GameObjects.Graphics> = new Map();
  private isLoading: boolean = true;

  constructor() {
    super({ key: "MapSelectionScene" });
  }

  async create(): Promise<void> {
    const { width, height } = this.scale;

    // Background
    this.add.rectangle(width / 2, height / 2, width, height, 0x0a0a1a, 1);

    // Title
    this.add.text(width / 2, 40, "SELECT MAP", {
      fontFamily: "Courier New",
      fontSize: "32px",
      color: "#00ff66",
      fontStyle: "bold",
    }).setOrigin(0.5);

    // Loading text
    const loadingText = this.add.text(width / 2, height / 2, "Loading maps...", {
      fontFamily: "Courier New",
      fontSize: "18px",
      color: "#888888",
    }).setOrigin(0.5);

    // Load map info
    await this.loadMapInfos();
    loadingText.destroy();
    this.isLoading = false;

    // Create UI elements
    this.createMapList(50, 80);
    this.createMapInfoPanel(width - 350, 80);
    this.createTeamSelection(width - 350, 350);
    this.createButtons(width / 2, height - 60);

    // Select first map
    this.selectMap(0);
  }

  private async loadMapInfos(): Promise<void> {
    this.mapInfos = [];

    for (const mapPath of AVAILABLE_MAPS) {
      try {
        const mapData = await MapLoader.loadFromUrl(mapPath);
        this.mapInfos.push({
          path: mapPath,
          name: mapData.mapName || mapPath.split("/").pop()?.replace(".map", "") || "Unknown",
          width: mapData.width,
          height: mapData.height,
          terrainType: mapData.terrainType,
          playerCount: mapData.playerCount,
        });
      } catch (error) {
        console.warn(`Failed to load map info for ${mapPath}:`, error);
      }
    }
  }

  private createMapList(x: number, y: number): void {
    this.mapListContainer = this.add.container(x, y);

    // List background
    const listWidth = 350;
    const listHeight = 400;
    const bg = this.add.graphics();
    bg.fillStyle(0x1a1a2e, 0.95);
    bg.fillRoundedRect(0, 0, listWidth, listHeight, 8);
    bg.lineStyle(2, 0x3a3a5e, 1);
    bg.strokeRoundedRect(0, 0, listWidth, listHeight, 8);
    this.mapListContainer.add(bg);

    // List header
    this.add.text(x + 15, y + 15, "AVAILABLE MAPS", {
      fontFamily: "Courier New",
      fontSize: "14px",
      color: "#888888",
    });

    // Map entries
    const entryHeight = 50;
    const startY = 45;

    this.mapInfos.forEach((mapInfo, index) => {
      const entryY = startY + index * entryHeight;

      // Entry background (interactive)
      const entry = this.add.graphics();
      entry.fillStyle(index === this.selectedMapIndex ? 0x2a4a2a : 0x1a1a2e, 1);
      entry.fillRoundedRect(10, entryY, listWidth - 20, entryHeight - 5, 4);
      this.mapListContainer.add(entry);

      // Map name
      const nameText = this.add.text(x + 20, y + entryY + 10, mapInfo.name, {
        fontFamily: "Courier New",
        fontSize: "14px",
        color: "#ffffff",
        fontStyle: "bold",
      });

      // Map size
      const sizeText = this.add.text(x + 20, y + entryY + 28, `${mapInfo.width}x${mapInfo.height} - ${mapInfo.playerCount} players`, {
        fontFamily: "Courier New",
        fontSize: "11px",
        color: "#888888",
      });

      // Terrain indicator
      const terrainColor = this.getTerrainColor(mapInfo.terrainType);
      const terrainDot = this.add.graphics();
      terrainDot.fillStyle(terrainColor, 1);
      terrainDot.fillCircle(listWidth - 25, entryY + entryHeight / 2 - 2, 8);
      this.mapListContainer.add(terrainDot);

      // Make entry interactive
      entry.setInteractive(
        new Phaser.Geom.Rectangle(10, entryY, listWidth - 20, entryHeight - 5),
        Phaser.Geom.Rectangle.Contains
      );

      entry.on("pointerover", () => {
        if (index !== this.selectedMapIndex) {
          entry.clear();
          entry.fillStyle(0x2a3a3a, 1);
          entry.fillRoundedRect(10, entryY, listWidth - 20, entryHeight - 5, 4);
        }
      });

      entry.on("pointerout", () => {
        entry.clear();
        entry.fillStyle(index === this.selectedMapIndex ? 0x2a4a2a : 0x1a1a2e, 1);
        entry.fillRoundedRect(10, entryY, listWidth - 20, entryHeight - 5, 4);
      });

      entry.on("pointerdown", () => {
        this.selectMap(index);
      });

      // Store reference for updating selection
      entry.setData("index", index);
      entry.setData("entryY", entryY);
    });
  }

  private getTerrainColor(terrainType: number): number {
    switch (terrainType) {
      case PlanetType.DESERT: return 0xc4a000;
      case PlanetType.ARCTIC: return 0xaaccff;
      case PlanetType.VOLCANIC: return 0xff4400;
      case PlanetType.CITY: return 0x888888;
      case PlanetType.JUNGLE: return 0x00aa00;
      default: return 0x888888;
    }
  }

  private createMapInfoPanel(x: number, y: number): void {
    this.mapInfoPanel = this.add.container(x, y);

    const panelWidth = 300;
    const panelHeight = 200;

    // Panel background
    const bg = this.add.graphics();
    bg.fillStyle(0x1a1a2e, 0.95);
    bg.fillRoundedRect(0, 0, panelWidth, panelHeight, 8);
    bg.lineStyle(2, 0x3a3a5e, 1);
    bg.strokeRoundedRect(0, 0, panelWidth, panelHeight, 8);
    this.mapInfoPanel.add(bg);

    // Header
    const header = this.add.text(15, 15, "MAP DETAILS", {
      fontFamily: "Courier New",
      fontSize: "14px",
      color: "#888888",
    });
    this.mapInfoPanel.add(header);
  }

  private updateMapInfoPanel(): void {
    // Clear previous info (keep background and header)
    while (this.mapInfoPanel.length > 2) {
      this.mapInfoPanel.removeAt(2, true);
    }

    const mapInfo = this.mapInfos[this.selectedMapIndex];
    if (!mapInfo) return;

    const x = 15;
    let y = 50;
    const lineHeight = 28;

    // Map name
    const nameLabel = this.add.text(x, y, "Name:", { fontFamily: "Courier New", fontSize: "12px", color: "#888888" });
    const nameValue = this.add.text(x + 80, y, mapInfo.name, { fontFamily: "Courier New", fontSize: "12px", color: "#ffffff" });
    this.mapInfoPanel.add([nameLabel, nameValue]);
    y += lineHeight;

    // Size
    const sizeLabel = this.add.text(x, y, "Size:", { fontFamily: "Courier New", fontSize: "12px", color: "#888888" });
    const sizeValue = this.add.text(x + 80, y, `${mapInfo.width} x ${mapInfo.height} tiles`, { fontFamily: "Courier New", fontSize: "12px", color: "#ffffff" });
    this.mapInfoPanel.add([sizeLabel, sizeValue]);
    y += lineHeight;

    // Terrain
    const terrainLabel = this.add.text(x, y, "Terrain:", { fontFamily: "Courier New", fontSize: "12px", color: "#888888" });
    const terrainName = PLANET_TYPE_STRINGS[mapInfo.terrainType as keyof typeof PLANET_TYPE_STRINGS] ?? "Unknown";
    const terrainValue = this.add.text(x + 80, y, terrainName.charAt(0).toUpperCase() + terrainName.slice(1), { fontFamily: "Courier New", fontSize: "12px", color: "#ffffff" });
    this.mapInfoPanel.add([terrainLabel, terrainValue]);
    y += lineHeight;

    // Players
    const playersLabel = this.add.text(x, y, "Players:", { fontFamily: "Courier New", fontSize: "12px", color: "#888888" });
    const playersValue = this.add.text(x + 80, y, `${mapInfo.playerCount}`, { fontFamily: "Courier New", fontSize: "12px", color: "#ffffff" });
    this.mapInfoPanel.add([playersLabel, playersValue]);
  }

  private createTeamSelection(x: number, y: number): void {
    const container = this.add.container(x, y);

    const panelWidth = 300;
    const panelHeight = 120;

    // Panel background
    const bg = this.add.graphics();
    bg.fillStyle(0x1a1a2e, 0.95);
    bg.fillRoundedRect(0, 0, panelWidth, panelHeight, 8);
    bg.lineStyle(2, 0x3a3a5e, 1);
    bg.strokeRoundedRect(0, 0, panelWidth, panelHeight, 8);
    container.add(bg);

    // Header
    this.add.text(x + 15, y + 15, "SELECT TEAM", {
      fontFamily: "Courier New",
      fontSize: "14px",
      color: "#888888",
    });

    // Team buttons
    const teams = [TeamType.RED, TeamType.BLUE, TeamType.GREEN, TeamType.YELLOW];
    const teamNames = ["RED", "BLUE", "GREEN", "YELLOW"];
    const buttonSize = 50;
    const startX = x + 30;
    const buttonY = y + 55;
    const spacing = 65;

    teams.forEach((team, index) => {
      const buttonX = startX + index * spacing;
      const teamColor = TEAM_COLORS[team] ?? 0x888888;

      const button = this.add.graphics();
      this.drawTeamButton(button, buttonX, buttonY, buttonSize, teamColor, team === this.selectedTeam);
      this.teamButtons.set(team, button);

      // Team label
      this.add.text(buttonX + buttonSize / 2, buttonY + buttonSize + 8, teamNames[index] ?? "", {
        fontFamily: "Courier New",
        fontSize: "10px",
        color: "#888888",
      }).setOrigin(0.5);

      // Make interactive
      button.setInteractive(
        new Phaser.Geom.Rectangle(buttonX, buttonY, buttonSize, buttonSize),
        Phaser.Geom.Rectangle.Contains
      );

      button.on("pointerdown", () => {
        this.selectTeam(team);
      });
    });
  }

  private drawTeamButton(graphics: Phaser.GameObjects.Graphics, x: number, y: number, size: number, color: number, selected: boolean): void {
    graphics.clear();

    // Background
    graphics.fillStyle(color, selected ? 1 : 0.5);
    graphics.fillRoundedRect(x, y, size, size, 8);

    // Border
    graphics.lineStyle(selected ? 3 : 2, selected ? 0xffffff : 0x666666, 1);
    graphics.strokeRoundedRect(x, y, size, size, 8);
  }

  private selectTeam(team: TeamType): void {
    this.selectedTeam = team;

    // Update all team buttons
    const teams = [TeamType.RED, TeamType.BLUE, TeamType.GREEN, TeamType.YELLOW];
    const buttonSize = 50;

    teams.forEach((t, index) => {
      const button = this.teamButtons.get(t);
      if (button) {
        const buttonX = this.mapInfoPanel.x + 30 + index * 65 - this.mapInfoPanel.x;
        const buttonY = 55;
        const teamColor = TEAM_COLORS[t] ?? 0x888888;
        // Recalculate position relative to team selection panel
        const panelX = this.scale.width - 350;
        this.drawTeamButton(button, panelX + 30 + index * 65, 350 + 55, buttonSize, teamColor, t === team);
      }
    });
  }

  private selectMap(index: number): void {
    this.selectedMapIndex = index;

    // Update map list visuals
    this.mapListContainer.each((child: Phaser.GameObjects.GameObject) => {
      if (child instanceof Phaser.GameObjects.Graphics && child.getData("index") !== undefined) {
        const i = child.getData("index") as number;
        const entryY = child.getData("entryY") as number;
        child.clear();
        child.fillStyle(i === index ? 0x2a4a2a : 0x1a1a2e, 1);
        child.fillRoundedRect(10, entryY, 330, 45, 4);
      }
    });

    // Update info panel
    this.updateMapInfoPanel();
  }

  private createButtons(x: number, y: number): void {
    // Back button
    this.createActionButton(x - 120, y, "BACK", () => {
      this.scene.start("MainMenuScene");
    }, 0x5a3a3a, 0x7a4a4a);

    // Start button
    this.createActionButton(x + 120, y, "START GAME", () => {
      const mapInfo = this.mapInfos[this.selectedMapIndex];
      if (mapInfo) {
        this.scene.start("GameScene", {
          mapPath: mapInfo.path,
          team: this.selectedTeam,
        });
      }
    }, 0x3a5a3a, 0x4a7a4a);
  }

  private createActionButton(x: number, y: number, text: string, onClick: () => void, color: number, hoverColor: number): void {
    const buttonWidth = 180;
    const buttonHeight = 45;

    const button = this.add.graphics();
    button.fillStyle(color, 1);
    button.fillRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
    button.lineStyle(2, hoverColor, 1);
    button.strokeRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);

    const buttonText = this.add.text(x, y, text, {
      fontFamily: "Courier New",
      fontSize: "16px",
      color: "#ffffff",
      fontStyle: "bold",
    }).setOrigin(0.5);

    button.setInteractive(
      new Phaser.Geom.Rectangle(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight),
      Phaser.Geom.Rectangle.Contains
    );

    button.on("pointerover", () => {
      button.clear();
      button.fillStyle(hoverColor, 1);
      button.fillRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
      button.lineStyle(2, 0xffffff, 0.5);
      button.strokeRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
    });

    button.on("pointerout", () => {
      button.clear();
      button.fillStyle(color, 1);
      button.fillRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
      button.lineStyle(2, hoverColor, 1);
      button.strokeRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
    });

    button.on("pointerdown", onClick);
  }
}
```
