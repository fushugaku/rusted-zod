```typescript
/**
 * GameSettings - Manages game configuration with localStorage persistence
 */

const STORAGE_KEY = "zod_settings";

export interface GameSettingsData {
  /** Master volume (0-100) */
  masterVolume: number;
  /** Sound effects volume (0-100) */
  soundVolume: number;
  /** Voice/announcements volume (0-100) */
  voiceVolume: number;
  /** Game speed multiplier (0.5, 1.0, 1.5, 2.0) */
  gameSpeed: number;
  /** Show health bars above units */
  showHealthBars: boolean;
  /** Show unit names on hover */
  showUnitNames: boolean;
  /** Minimap size (small, medium, large) */
  minimapSize: "small" | "medium" | "large";
}

const DEFAULT_SETTINGS: GameSettingsData = {
  masterVolume: 80,
  soundVolume: 100,
  voiceVolume: 100,
  gameSpeed: 1.0,
  showHealthBars: true,
  showUnitNames: true,
  minimapSize: "medium",
};

/**
 * Current settings instance
 */
let currentSettings: GameSettingsData = { ...DEFAULT_SETTINGS };

/**
 * Load settings from localStorage
 */
export function loadSettings(): GameSettingsData {
  try {
    const saved = localStorage.getItem(STORAGE_KEY);
    if (saved) {
      const parsed = JSON.parse(saved) as Partial<GameSettingsData>;
      // Merge with defaults to handle new settings fields
      currentSettings = { ...DEFAULT_SETTINGS, ...parsed };
    } else {
      currentSettings = { ...DEFAULT_SETTINGS };
    }
  } catch (error) {
    console.warn("Failed to load settings from localStorage:", error);
    currentSettings = { ...DEFAULT_SETTINGS };
  }
  return currentSettings;
}

/**
 * Save settings to localStorage
 */
export function saveSettings(settings: GameSettingsData): void {
  try {
    currentSettings = { ...settings };
    localStorage.setItem(STORAGE_KEY, JSON.stringify(settings));
  } catch (error) {
    console.warn("Failed to save settings to localStorage:", error);
  }
}

/**
 * Get current settings (without loading from storage)
 */
export function getSettings(): GameSettingsData {
  return { ...currentSettings };
}

/**
 * Reset settings to defaults
 */
export function resetSettings(): GameSettingsData {
  currentSettings = { ...DEFAULT_SETTINGS };
  saveSettings(currentSettings);
  return currentSettings;
}

/**
 * Update a single setting
 */
export function updateSetting<K extends keyof GameSettingsData>(
  key: K,
  value: GameSettingsData[K]
): void {
  currentSettings[key] = value;
  saveSettings(currentSettings);
}

/**
 * Calculate effective volume (master * specific)
 */
export function getEffectiveVolume(volumeType: "sound" | "voice"): number {
  const master = currentSettings.masterVolume / 100;
  const specific = volumeType === "sound"
    ? currentSettings.soundVolume / 100
    : currentSettings.voiceVolume / 100;
  return master * specific;
}

/**
 * Game speed presets
 */
export const GAME_SPEED_PRESETS = [
  { value: 0.5, label: "Slow (0.5x)" },
  { value: 1.0, label: "Normal (1.0x)" },
  { value: 1.5, label: "Fast (1.5x)" },
  { value: 2.0, label: "Faster (2.0x)" },
];

// Initialize settings on module load
loadSettings();
```

```typescript
import Phaser from "phaser";
import {
  loadSettings,
  saveSettings,
  resetSettings,
  GAME_SPEED_PRESETS,
  type GameSettingsData,
} from "@/config/GameSettings";

export class SettingsScene extends Phaser.Scene {
  private settings!: GameSettingsData;
  private sliders: Map<string, { bar: Phaser.GameObjects.Graphics; handle: Phaser.GameObjects.Graphics; valueText: Phaser.GameObjects.Text }> = new Map();
  private speedButtons: Map<number, Phaser.GameObjects.Graphics> = new Map();

  constructor() {
    super({ key: "SettingsScene" });
  }

  create(): void {
    const { width, height } = this.scale;

    // Load current settings
    this.settings = loadSettings();

    // Background
    this.add.rectangle(width / 2, height / 2, width, height, 0x0a0a1a, 1);

    // Title
    this.add.text(width / 2, 50, "SETTINGS", {
      fontFamily: "Courier New",
      fontSize: "36px",
      color: "#00ff66",
      fontStyle: "bold",
    }).setOrigin(0.5);

    // Settings panel
    const panelX = width / 2 - 250;
    const panelY = 100;
    this.createSettingsPanel(panelX, panelY);

    // Buttons
    this.createButtons(width / 2, height - 70);
  }

  private createSettingsPanel(x: number, y: number): void {
    const panelWidth = 500;
    const panelHeight = 380;

    // Panel background
    const panel = this.add.graphics();
    panel.fillStyle(0x1a1a2e, 0.95);
    panel.fillRoundedRect(x, y, panelWidth, panelHeight, 8);
    panel.lineStyle(2, 0x3a3a5e, 1);
    panel.strokeRoundedRect(x, y, panelWidth, panelHeight, 8);

    let currentY = y + 30;
    const labelX = x + 30;
    const controlX = x + 180;

    // Audio section header
    this.add.text(labelX, currentY, "AUDIO", {
      fontFamily: "Courier New",
      fontSize: "16px",
      color: "#00ff66",
      fontStyle: "bold",
    });
    currentY += 35;

    // Master volume
    this.createSlider("masterVolume", "Master Volume", labelX, controlX, currentY, this.settings.masterVolume);
    currentY += 50;

    // Sound effects volume
    this.createSlider("soundVolume", "Sound Effects", labelX, controlX, currentY, this.settings.soundVolume);
    currentY += 50;

    // Voice volume
    this.createSlider("voiceVolume", "Announcements", labelX, controlX, currentY, this.settings.voiceVolume);
    currentY += 60;

    // Game section header
    this.add.text(labelX, currentY, "GAME", {
      fontFamily: "Courier New",
      fontSize: "16px",
      color: "#00ff66",
      fontStyle: "bold",
    });
    currentY += 35;

    // Game speed
    this.add.text(labelX, currentY, "Game Speed", {
      fontFamily: "Courier New",
      fontSize: "14px",
      color: "#cccccc",
    });
    this.createSpeedSelector(controlX, currentY);
    currentY += 50;

    // Toggles section
    this.add.text(labelX, currentY, "DISPLAY", {
      fontFamily: "Courier New",
      fontSize: "16px",
      color: "#00ff66",
      fontStyle: "bold",
    });
    currentY += 35;

    // Show health bars toggle
    this.createToggle("showHealthBars", "Show Health Bars", labelX, controlX, currentY, this.settings.showHealthBars);
  }

  private createSlider(key: string, label: string, labelX: number, controlX: number, y: number, initialValue: number): void {
    // Label
    this.add.text(labelX, y, label, {
      fontFamily: "Courier New",
      fontSize: "14px",
      color: "#cccccc",
    });

    const sliderWidth = 200;
    const sliderHeight = 8;

    // Slider track
    const track = this.add.graphics();
    track.fillStyle(0x333333, 1);
    track.fillRoundedRect(controlX, y + 4, sliderWidth, sliderHeight, 4);

    // Slider fill (bar)
    const bar = this.add.graphics();
    this.drawSliderBar(bar, controlX, y + 4, sliderWidth, sliderHeight, initialValue);

    // Slider handle
    const handleX = controlX + (initialValue / 100) * sliderWidth;
    const handle = this.add.graphics();
    handle.fillStyle(0x00ff66, 1);
    handle.fillCircle(handleX, y + 8, 10);
    handle.lineStyle(2, 0xffffff, 0.5);
    handle.strokeCircle(handleX, y + 8, 10);

    // Value text
    const valueText = this.add.text(controlX + sliderWidth + 15, y, `${initialValue}%`, {
      fontFamily: "Courier New",
      fontSize: "14px",
      color: "#ffffff",
    });

    // Store references
    this.sliders.set(key, { bar, handle, valueText });

    // Make track interactive for clicking
    track.setInteractive(
      new Phaser.Geom.Rectangle(controlX, y, sliderWidth, 16),
      Phaser.Geom.Rectangle.Contains
    );

    // Handle drag
    handle.setInteractive(
      new Phaser.Geom.Circle(handleX, y + 8, 12),
      Phaser.Geom.Circle.Contains
    );
    this.input.setDraggable(handle);

    handle.on("drag", (_pointer: Phaser.Input.Pointer, dragX: number) => {
      const clampedX = Phaser.Math.Clamp(dragX, controlX, controlX + sliderWidth);
      const value = Math.round(((clampedX - controlX) / sliderWidth) * 100);
      this.updateSlider(key, controlX, y, sliderWidth, sliderHeight, value);
    });

    track.on("pointerdown", (pointer: Phaser.Input.Pointer) => {
      const value = Math.round(((pointer.x - controlX) / sliderWidth) * 100);
      const clampedValue = Phaser.Math.Clamp(value, 0, 100);
      this.updateSlider(key, controlX, y, sliderWidth, sliderHeight, clampedValue);
    });
  }

  private drawSliderBar(bar: Phaser.GameObjects.Graphics, x: number, y: number, width: number, height: number, value: number): void {
    bar.clear();
    bar.fillStyle(0x00aa44, 1);
    bar.fillRoundedRect(x, y, (value / 100) * width, height, 4);
  }

  private updateSlider(key: string, x: number, y: number, width: number, height: number, value: number): void {
    const slider = this.sliders.get(key);
    if (!slider) return;

    // Update bar
    this.drawSliderBar(slider.bar, x, y + 4, width, height, value);

    // Update handle position
    const handleX = x + (value / 100) * width;
    slider.handle.clear();
    slider.handle.fillStyle(0x00ff66, 1);
    slider.handle.fillCircle(handleX, y + 8, 10);
    slider.handle.lineStyle(2, 0xffffff, 0.5);
    slider.handle.strokeCircle(handleX, y + 8, 10);

    // Update value text
    slider.valueText.setText(`${value}%`);

    // Update settings
    (this.settings as Record<string, number | boolean | string>)[key] = value;
  }

  private createSpeedSelector(x: number, y: number): void {
    const buttonWidth = 70;
    const buttonHeight = 28;
    const spacing = 5;

    GAME_SPEED_PRESETS.forEach((preset, index) => {
      const buttonX = x + index * (buttonWidth + spacing);
      const isSelected = this.settings.gameSpeed === preset.value;

      const button = this.add.graphics();
      this.drawSpeedButton(button, buttonX, y - 5, buttonWidth, buttonHeight, isSelected);
      this.speedButtons.set(preset.value, button);

      // Label
      const label = preset.value === 1.0 ? "1x" : `${preset.value}x`;
      this.add.text(buttonX + buttonWidth / 2, y + buttonHeight / 2 - 5, label, {
        fontFamily: "Courier New",
        fontSize: "12px",
        color: isSelected ? "#00ff66" : "#888888",
      }).setOrigin(0.5);

      // Interactive
      button.setInteractive(
        new Phaser.Geom.Rectangle(buttonX, y - 5, buttonWidth, buttonHeight),
        Phaser.Geom.Rectangle.Contains
      );

      button.on("pointerdown", () => {
        this.selectSpeed(preset.value);
      });
    });
  }

  private drawSpeedButton(button: Phaser.GameObjects.Graphics, x: number, y: number, width: number, height: number, selected: boolean): void {
    button.clear();
    button.fillStyle(selected ? 0x2a4a2a : 0x1a1a2e, 1);
    button.fillRoundedRect(x, y, width, height, 4);
    button.lineStyle(2, selected ? 0x00ff66 : 0x3a3a5e, 1);
    button.strokeRoundedRect(x, y, width, height, 4);
  }

  private selectSpeed(speed: number): void {
    this.settings.gameSpeed = speed;

    // Update all speed buttons
    GAME_SPEED_PRESETS.forEach((preset) => {
      const button = this.speedButtons.get(preset.value);
      if (button) {
        // Need to recalculate position - use fixed values
        const x = this.scale.width / 2 - 250 + 180;
        const buttonWidth = 70;
        const spacing = 5;
        const index = GAME_SPEED_PRESETS.findIndex(p => p.value === preset.value);
        const buttonX = x + index * (buttonWidth + spacing);
        this.drawSpeedButton(button, buttonX, 100 + 30 + 35 + 50 + 50 + 60 + 35 - 5, buttonWidth, 28, preset.value === speed);
      }
    });
  }

  private createToggle(key: string, label: string, labelX: number, controlX: number, y: number, initialValue: boolean): void {
    // Label
    this.add.text(labelX, y, label, {
      fontFamily: "Courier New",
      fontSize: "14px",
      color: "#cccccc",
    });

    const toggleWidth = 50;
    const toggleHeight = 24;

    const toggle = this.add.graphics();
    this.drawToggle(toggle, controlX, y - 2, toggleWidth, toggleHeight, initialValue);

    toggle.setInteractive(
      new Phaser.Geom.Rectangle(controlX, y - 2, toggleWidth, toggleHeight),
      Phaser.Geom.Rectangle.Contains
    );

    toggle.on("pointerdown", () => {
      const newValue = !(this.settings as Record<string, boolean>)[key];
      (this.settings as Record<string, boolean>)[key] = newValue;
      this.drawToggle(toggle, controlX, y - 2, toggleWidth, toggleHeight, newValue);
    });
  }

  private drawToggle(graphics: Phaser.GameObjects.Graphics, x: number, y: number, width: number, height: number, enabled: boolean): void {
    graphics.clear();

    // Track
    graphics.fillStyle(enabled ? 0x2a4a2a : 0x333333, 1);
    graphics.fillRoundedRect(x, y, width, height, height / 2);
    graphics.lineStyle(2, enabled ? 0x00ff66 : 0x555555, 1);
    graphics.strokeRoundedRect(x, y, width, height, height / 2);

    // Handle
    const handleX = enabled ? x + width - height / 2 - 2 : x + height / 2 + 2;
    graphics.fillStyle(enabled ? 0x00ff66 : 0x888888, 1);
    graphics.fillCircle(handleX, y + height / 2, height / 2 - 4);
  }

  private createButtons(x: number, y: number): void {
    // Reset button
    this.createActionButton(x - 150, y, "RESET", () => {
      this.settings = resetSettings();
      // Recreate scene to refresh UI
      this.scene.restart();
    }, 0x5a3a3a, 0x7a4a4a);

    // Back button (saves and returns)
    this.createActionButton(x + 150, y, "SAVE & BACK", () => {
      saveSettings(this.settings);
      this.scene.start("MainMenuScene");
    }, 0x3a5a3a, 0x4a7a4a);
  }

  private createActionButton(x: number, y: number, text: string, onClick: () => void, color: number, hoverColor: number): void {
    const buttonWidth = 150;
    const buttonHeight = 40;

    const button = this.add.graphics();
    button.fillStyle(color, 1);
    button.fillRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
    button.lineStyle(2, hoverColor, 1);
    button.strokeRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);

    this.add.text(x, y, text, {
      fontFamily: "Courier New",
      fontSize: "14px",
      color: "#ffffff",
      fontStyle: "bold",
    }).setOrigin(0.5);

    button.setInteractive(
      new Phaser.Geom.Rectangle(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight),
      Phaser.Geom.Rectangle.Contains
    );

    button.on("pointerover", () => {
      button.clear();
      button.fillStyle(hoverColor, 1);
      button.fillRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
      button.lineStyle(2, 0xffffff, 0.5);
      button.strokeRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
    });

    button.on("pointerout", () => {
      button.clear();
      button.fillStyle(color, 1);
      button.fillRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
      button.lineStyle(2, hoverColor, 1);
      button.strokeRoundedRect(x - buttonWidth / 2, y - buttonHeight / 2, buttonWidth, buttonHeight, 8);
    });

    button.on("pointerdown", onClick);
  }
}
```
