   ```typescript
    * Add a robot to the garrison
    * From bfort.cpp CanEnterFort check + garrison logic
    */
   public addToGarrison(
     refId: number,
     robotType: RobotType,
     health: number,
     maxHealth: number,
     grenades: number
   ): boolean {
     // Check if can accept more
     if (this.garrisonedUnits.length >= Fort.MAX_GARRISON) {
       return false;
     }

     // Check if fort is owned by different team (can't enter enemy fort)
     // From bfort.cpp: CanEnterFort - team == owner returns false
     // Actually, the check is: can enter if NOT owner (to capture)
     // For our garrison mechanic, only same-team robots garrison

     this.garrisonedUnits.push({
       refId,
       robotType,
       health,
       maxHealth,
       grenades,
       lastAttackTime: 0,
     });

     console.log(`Fort ${this.refId}: Robot ${refId} (${robotType}) garrisoned. Total: ${this.garrisonedUnits.length}`);
     return true;
   }

   /**
    * Remove a robot from garrison (exit or death)
    */
   public removeFromGarrison(refId: number): GarrisonedUnit | null {
     const index = this.garrisonedUnits.findIndex(u => u.refId === refId);
     if (index === -1) return null;

     const unit = this.garrisonedUnits.splice(index, 1)[0];
     console.log(`Fort ${this.refId}: Robot ${refId} left garrison. Remaining: ${this.garrisonedUnits.length}`);
     return unit ?? null;
   }

   /**
    * Get all garrisoned units
    */
   public getGarrisonedUnits(): readonly GarrisonedUnit[] {
     return this.garrisonedUnits;
   }

   /**
    * Check if garrison has room
    */
   public canAcceptGarrison(): boolean {
     return this.garrisonedUnits.length < Fort.MAX_GARRISON;
   }

   /**
    * Get garrison count
    */
   public getGarrisonCount(): number {
     return this.garrisonedUnits.length;
   }
   ```

   ```typescript
    * Can this fort attack? Requires at least one garrisoned unit

   /**
    * Get the best attack range from garrisoned units
    * Uses the longest range unit's attack radius
    */
   public getAttackRange(): number {
     if (this.garrisonedUnits.length === 0) return 0;

     let maxRange = 0;
     for (const unit of this.garrisonedUnits) {
       const stats = CombatSystem.getCombatStats(ObjectType.OBJECT_ROBOT, unit.robotType);
       if (stats && stats.attackRadius > maxRange) {
         maxRange = stats.attackRadius;
       }
     }
     return maxRange;
   }
   ```

1. **Add fort attack method to CombatSystem**:
   ```typescript
   /**
    * Process an attack from a fort's garrisoned units
    * Each garrisoned unit fires independently based on their attack speed
    */
   public processFortAttack(
     fort: Fort,
     targetRefId: number,
     targetX: number,
     targetY: number,
     targetHealth: number,
     targetMaxHealth: number,
     time: number
   ): { totalDamage: number; hits: number } {
     const garrison = fort.getGarrisonedUnits();
     if (garrison.length === 0) {
       return { totalDamage: 0, hits: 0 };
     }

     let totalDamage = 0;
     let hits = 0;

     // Fort center as attack origin
     const fortX = fort.x + fort.objectWidth / 2;
     const fortY = fort.y + fort.objectHeight / 2;
     const distance = Phaser.Math.Distance.Between(fortX, fortY, targetX, targetY);

     for (const unit of garrison) {
       const stats = CombatSystem.getCombatStats(ObjectType.OBJECT_ROBOT, unit.robotType);
       if (!stats) continue;

       // Check if in range
       if (distance > stats.attackRadius) continue;

       // Check attack cooldown
       if (time - unit.lastAttackTime < stats.attackSpeed) continue;

       // Update last attack time
       unit.lastAttackTime = time;

       // Roll for hit
       if (Math.random() > stats.damageChance) {
         console.log(`Fort ${fort.refId}: Unit ${unit.refId} missed`);
         continue;
       }

       // Calculate damage
       const damageAmount = stats.damage * targetMaxHealth;
       totalDamage += damageAmount;
       hits++;

       // Handle missile vs instant damage
       if (stats.damageIsMissile) {
         // Create missile from fort position
         const travelTime = distance / stats.missileSpeed * 1000;
         const missile: DamageMissile = {
           x: targetX,
           y: targetY,
           originX: fortX,
           originY: fortY,
           damage: damageAmount,
           radius: stats.damageRadius,
           team: fort.team,
           attackerRefId: fort.refId, // Fort is the attacker for tracking
           targetRefId: targetRefId,
           explodeTime: time + travelTime,
           playerGiven: true,
           missileType: this.getMissileTypeForRobot(unit.robotType),
         };
         this.missiles.push(missile);
         this.onMissileFired?.(missile);
       } else {
         // Instant damage
         this.applyDamageToUnit?.(targetRefId, damageAmount, fort.refId);
         this.onDamageDealt?.(targetRefId, damageAmount, fort.refId);
       }

       // Trigger attack effect
       this.onAttackEffect?.(fort.refId, targetRefId,
         this.getMissileTypeForRobot(unit.robotType));

       console.log(`Fort ${fort.refId}: Unit ${unit.refId} hit target for ${damageAmount.toFixed(0)} damage`);
     }

     return { totalDamage, hits };
   }

   /**
    * Get missile type for a robot type
    */
   private getMissileTypeForRobot(robotType: RobotType): MissileType {
     switch (robotType) {
       case RobotType.GRUNT: return MissileType.BULLET;
       case RobotType.PSYCHO: return MissileType.BULLET;
       case RobotType.SNIPER: return MissileType.BULLET;
       case RobotType.TOUGH: return MissileType.ROCKET;
       case RobotType.PYRO: return MissileType.FLAME;
       case RobotType.LASER: return MissileType.LASER;
       default: return MissileType.BULLET;
     }
   }
   ```

2. **Add fort attack command to Fort class**:
   ```typescript
   /**
    * Command garrisoned units to attack a target
    */
   public attackTarget(
     targetRefId: number,
     combatSystem: CombatSystem,
     getTargetInfo: () => { x: number; y: number; health: number; maxHealth: number } | null
   ): void {
     if (!this.canAttack()) return;

     this.attackTarget = targetRefId;

     // Store target getter for update loop
     this.getTargetInfo = getTargetInfo;
   }

   /**
    * Stop attacking
    */
   public stopAttack(): void {
     this.attackTarget = null;
     this.getTargetInfo = null;
   }

   /**
    * Process attack in update loop
    */
   public processAttack(time: number, combatSystem: CombatSystem): void {
     if (!this.attackTarget || !this.getTargetInfo) return;

     const targetInfo = this.getTargetInfo();
     if (!targetInfo) {
       // Target gone
       this.stopAttack();
       return;
     }

     // Check if target in range
     const fortX = this.x + this.objectWidth / 2;
     const fortY = this.y + this.objectHeight / 2;
     const distance = Phaser.Math.Distance.Between(
       fortX, fortY, targetInfo.x, targetInfo.y
     );

     if (distance > this.getAttackRange()) {
       // Out of range, stop attacking
       this.stopAttack();
       return;
     }

     // Process attack through combat system
     combatSystem.processFortAttack(
       this,
       this.attackTarget,
       targetInfo.x,
       targetInfo.y,
       targetInfo.health,
       targetInfo.maxHealth,
       time
     );

     // Check if target destroyed
     if (targetInfo.health <= 0) {
       this.stopAttack();
     }
   }
   ```

   ```typescript
     }
   }
   ```

   ```typescript
   // In Fort.ts update or in a garrison AI method
   public updateGarrisonCombat(
     time: number,
     getEnemiesInRange: () => Array<{ refId: number; x: number; y: number; health: number; maxHealth: number }>
   ): void {
     if (!this.canAttack()) return;
     if (this.attackTarget) return; // Already attacking

     // Find enemies in range
     const enemies = getEnemiesInRange();
     if (enemies.length === 0) return;

     // Attack nearest enemy
     const fortX = this.x + this.objectWidth / 2;
     const fortY = this.y + this.objectHeight / 2;

     let nearest = enemies[0];
     let nearestDist = Phaser.Math.Distance.Between(fortX, fortY, nearest.x, nearest.y);

     for (const enemy of enemies) {
       const dist = Phaser.Math.Distance.Between(fortX, fortY, enemy.x, enemy.y);
       if (dist < nearestDist) {
         nearest = enemy;
         nearestDist = dist;
       }
     }

     if (nearest && nearestDist <= this.getAttackRange()) {
       this.attackTarget = nearest.refId;
     }
   }
   ```
