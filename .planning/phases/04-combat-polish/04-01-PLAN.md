   import { ItemType, TeamType, ObjectType } from "@/types";
   import { GameObject } from "../GameObject";

   /**
    * GrenadeBox - Pickupable grenade container
    * From source/ogrenades.cpp
    */
   export class GrenadeBox extends GameObject {
     private grenadeAmount: number;
     private static readonly DEFAULT_GRENADES_PER_BOX = 5;

     constructor(scene: Phaser.Scene, state: ItemState) {
       super(scene, state);
       this.objectType = ObjectType.OBJECT_ITEM;
       this.objectId = ItemType.GRENADES;

       // From ogrenades.cpp: grenade_amount = zsettings->grenades_per_box
       this.grenadeAmount = state.count ?? GrenadeBox.DEFAULT_GRENADES_PER_BOX;

       // Size from original: width_pix = 16, height_pix = 16
       this.objectWidth = 16;
       this.objectHeight = 16;

       this.createVisuals();
     }

     protected createVisuals(): void {
       // Fallback graphics - yellow/green box with "G" label
       const graphics = this.scene.add.graphics();
       graphics.fillStyle(0x556b2f, 1); // Olive green
       graphics.fillRect(-8, -8, 16, 16);
       graphics.fillStyle(0xffff00, 1);
       graphics.fillCircle(0, 0, 4); // Yellow center
       this.add(graphics);
     }

     /**
      * Get grenade count in this box
      */
     public getGrenadeAmount(): number {
       return this.grenadeAmount;
     }

     /**
      * Pickup by a robot - returns amount picked up, destroys box
      */
     public pickup(): number {
       const amount = this.grenadeAmount;
       this.destroy();
       return amount;
     }

     /**
      * Can be picked up by this team
      * From ogrenades.cpp - grenades can be picked up by any team
      */
     public canPickup(_team: TeamType): boolean {
       return true;
     }
   }
   ```

   /**
    * Throw a grenade at target position
    * From zrobot.cpp FireMissile
    */
   public throwGrenade(targetX: number, targetY: number, combatSystem: CombatSystem): void {
     if (!this.canThrowGrenades()) return;

     // Face the target
     const dx = targetX - this.x;
     const dy = targetY - this.y;
     this.updateRotationFromDirection(dx, dy);

     // Decrement grenade count
     this.grenadeAmount--;

     // Play throw animation
     this.setAnimation(RobotAnimation.THROW);

     // Create grenade missile via combat system
     // From zrobot.cpp: dist_time = dist / zsettings->grenade_missile_speed
     combatSystem.createGrenadeMissile(
       this.refId,
       this.x,
       this.y,
       targetX,
       targetY,
       this.team
     );
   }
   ```

2. **Add createGrenadeMissile() to CombatSystem.ts**:
   ```typescript
   /**
    * Create a grenade projectile
    * From zrobot.cpp FireMissile and GRENADE_SETTINGS
    */
   public createGrenadeMissile(
     attackerRefId: number,
     originX: number,
     originY: number,
     targetX: number,
     targetY: number,
     team: TeamType
   ): DamageMissile {
     // Add randomness to target - from createMissile: randomOffset = 24 for grenades
     const randomOffset = 24;
     const finalX = targetX + (Math.random() * randomOffset * 2 - randomOffset);
     const finalY = targetY + (Math.random() * randomOffset * 2 - randomOffset);

     const distance = Phaser.Math.Distance.Between(originX, originY, finalX, finalY);
     const travelTime = (distance / GRENADE_SETTINGS.missileSpeed) * 1000;

     const missile: DamageMissile = {
       x: finalX,
       y: finalY,
       originX,
       originY,
       damage: GRENADE_SETTINGS.damage,
       radius: GRENADE_SETTINGS.damageRadius,
       team,
       attackerRefId,
       targetRefId: -1, // Area target, not specific unit
       explodeTime: this.gameTime + travelTime,
       playerGiven: true,
       missileType: MissileType.GRENADE,
     };

     this.missiles.push(missile);
     this.onMissileFired?.(missile);

     return missile;
   }
   ```

   public getGrenadeMissilePosition(
     missile: DamageMissile,
     time: number
   ): { x: number; y: number; height: number } {
     const progress = this.getMissileProgress(missile, time);
     const basePos = {
       x: missile.originX + (missile.x - missile.originX) * progress,
       y: missile.originY + (missile.y - missile.originY) * progress,
     };

     // Parabolic arc - peaks at 50% progress
     // height = 4 * maxHeight * progress * (1 - progress)
     const maxHeight = 30; // pixels
     const height = 4 * maxHeight * progress * (1 - progress);

     return { ...basePos, height };
   }
