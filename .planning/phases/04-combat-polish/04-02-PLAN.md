5. **Add driver damage method**:
   ```typescript
   /**
    * Apply damage to driver
    * Returns true if driver was killed
    */
   public damageDriver(damage: number): boolean {
     if (!this.hasDriver || this.driverHealth <= 0) {
       return false;
     }

     this.driverHealth = Math.max(0, this.driverHealth - damage);

     if (this.driverHealth <= 0) {
       this.killDriver();
       return true;
     }
     return false;
   }

   /**
    * Kill the driver - vehicle becomes driverless
    */
   protected killDriver(): void {
     this.hasDriver = false;
     this.driverType = null;
     this.driverHealth = 0;
     this.driverMaxHealth = 0;

     // Emit event for sound/visual effects
     this.scene.events.emit('driverKilled', {
       vehicleRefId: this.refId,
       x: this.x,
       y: this.y,
     });

     console.log(`Vehicle ${this.refId} driver killed - now driverless`);
   }
   ```

       this.onAttackEffect?.(

       );

       // Don't deal normal vehicle damage when sniping driver
       return;

     // Snipe roll failed - deal normal vehicle damage below
     console.log(`Combat: Sniper ${attack.attackerRefId} failed to hit driver, dealing vehicle damage`);
