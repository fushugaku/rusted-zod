1. **Create `client/src/effects/RepairEffect.ts`**:
   ```typescript
   import Phaser from "phaser";

   /**
    * Spark particle for repair effect
    * From edeathsparks.cpp - small orange/yellow particles
    */
   interface Spark {
     x: number;
     y: number;
     vx: number;
     vy: number;
     life: number;
     maxLife: number;
     size: number;
     color: number;
   }

   /**
    * RepairEffect - Visual feedback during repair actions
    * Shows sparks + wrench animation at repair location
    *
    * From brepair.cpp:
    * - smoke_stack animation when repairing
    * - Uses EDeathSparks-style particles
    */
   export class RepairEffect extends Phaser.GameObjects.Container {
     private sparks: Spark[] = [];
     private sparkGraphics: Phaser.GameObjects.Graphics;
     private wrenchSprite: Phaser.GameObjects.Graphics;

     // Effect timing
     private sparkSpawnTimer: number = 0;
     private wrenchRotation: number = 0;
     private active: boolean = true;

     // Configuration
     private static readonly SPARK_SPAWN_INTERVAL = 50; // ms between spark spawns
     private static readonly MAX_SPARKS = 20;
     private static readonly SPARK_LIFETIME = 500; // ms
     private static readonly SPARK_SPEED = 50; // pixels/sec
     private static readonly WRENCH_ROTATION_SPEED = 180; // degrees/sec

     // Colors from original EDeathSparks - orange/yellow sparks
     private static readonly SPARK_COLORS = [
       0xff8800, // Orange
       0xffaa00, // Light orange
       0xffcc00, // Yellow-orange
       0xffff00, // Yellow
       0xff6600, // Dark orange
       0xffee00, // Bright yellow
     ];

     constructor(scene: Phaser.Scene, x: number, y: number) {
       super(scene, x, y);

       // Create graphics for sparks
       this.sparkGraphics = scene.add.graphics();
       this.add(this.sparkGraphics);

       // Create wrench indicator
       this.wrenchSprite = scene.add.graphics();
       this.add(this.wrenchSprite);
       this.drawWrench();

       scene.add.existing(this);
     }

     /**
      * Draw wrench icon
      */
     private drawWrench(): void {
       this.wrenchSprite.clear();

       // Simple wrench shape using lines
       this.wrenchSprite.lineStyle(2, 0x888888, 1);

       // Handle
       this.wrenchSprite.lineBetween(-8, 0, 4, 0);

       // Head (open end)
       this.wrenchSprite.lineBetween(4, -4, 8, -4);
       this.wrenchSprite.lineBetween(4, 4, 8, 4);
       this.wrenchSprite.lineBetween(4, -4, 4, 4);
     }

     /**
      * Spawn a new spark particle
      */
     private spawnSpark(): void {
       if (this.sparks.length >= RepairEffect.MAX_SPARKS) return;

       // Random direction upward (repair sparks go up and out)
       const angle = -Math.PI / 2 + (Math.random() - 0.5) * Math.PI; // -180 to 0 degrees
       const speed = RepairEffect.SPARK_SPEED * (0.5 + Math.random() * 0.5);

       const spark: Spark = {
         x: (Math.random() - 0.5) * 10, // Spawn area
         y: (Math.random() - 0.5) * 10,
         vx: Math.cos(angle) * speed,
         vy: Math.sin(angle) * speed,
         life: RepairEffect.SPARK_LIFETIME,
         maxLife: RepairEffect.SPARK_LIFETIME,
         size: 2 + Math.random() * 2,
         color: RepairEffect.SPARK_COLORS[
           Math.floor(Math.random() * RepairEffect.SPARK_COLORS.length)
         ] ?? 0xff8800,
       };

       this.sparks.push(spark);
     }

     /**
      * Update effect
      */
     public update(time: number, delta: number): void {
       if (!this.active) return;

       // Spawn new sparks
       this.sparkSpawnTimer += delta;
       while (this.sparkSpawnTimer >= RepairEffect.SPARK_SPAWN_INTERVAL) {
         this.sparkSpawnTimer -= RepairEffect.SPARK_SPAWN_INTERVAL;
         this.spawnSpark();
       }

       // Update sparks
       const deltaSeconds = delta / 1000;
       for (let i = this.sparks.length - 1; i >= 0; i--) {
         const spark = this.sparks[i];
         if (!spark) continue;

         // Move
         spark.x += spark.vx * deltaSeconds;
         spark.y += spark.vy * deltaSeconds;

         // Gravity (slight downward pull)
         spark.vy += 100 * deltaSeconds;

         // Age
         spark.life -= delta;

         // Remove dead sparks
         if (spark.life <= 0) {
           this.sparks.splice(i, 1);
         }
       }

       // Rotate wrench
       this.wrenchRotation += RepairEffect.WRENCH_ROTATION_SPEED * deltaSeconds;
       this.wrenchSprite.setRotation(this.wrenchRotation * Math.PI / 180);

       // Render sparks
       this.renderSparks();
     }

     /**
      * Render spark particles
      */
     private renderSparks(): void {
       this.sparkGraphics.clear();

       for (const spark of this.sparks) {
         // Fade out as life decreases
         const alpha = spark.life / spark.maxLife;

         this.sparkGraphics.fillStyle(spark.color, alpha);
         this.sparkGraphics.fillCircle(spark.x, spark.y, spark.size * alpha);
       }
     }

     /**
      * Stop the effect (will fade out remaining sparks)
      */
     public stop(): void {
       this.active = false;
     }

     /**
      * Check if effect is complete (stopped and all sparks gone)
      */
     public isComplete(): boolean {
       return !this.active && this.sparks.length === 0;
     }

     /**
      * Destroy the effect
      */
     public override destroy(): void {
       this.sparks = [];
       this.sparkGraphics.destroy();
       this.wrenchSprite.destroy();
       super.destroy();
     }
   }
   ```

   ```typescript
   import { RepairEffect } from '@/effects';

   export class Building extends GameObject {
     // Repair effect when building self-repairs
     protected repairEffect: RepairEffect | null = null;
     protected isBeingRepaired: boolean = false;

     /**
      * Start repair effect (building self-repair or external repair)
      */
     public startRepairVisual(): void {
       if (this.repairEffect) return; // Already active

       this.repairEffect = new RepairEffect(
         this.scene,
         this.x + this.objectWidth / 2,
         this.y + this.objectHeight / 2
       );
       this.isBeingRepaired = true;
     }

     /**
      * Stop repair effect
      */
     public stopRepairVisual(): void {
       if (this.repairEffect) {
         this.repairEffect.stop();
         // Effect will self-destruct when complete
       }
       this.isBeingRepaired = false;
     }

     /**
      * Update repair visual
      */
     protected updateRepairVisual(time: number, delta: number): void {
       if (this.repairEffect) {
         this.repairEffect.update(time, delta);

         if (this.repairEffect.isComplete()) {
           this.repairEffect.destroy();
           this.repairEffect = null;
         }
       }
     }

     public override update(time: number, delta: number): void {
       super.update(time, delta);
       this.updateRepairVisual(time, delta);
     }
   }
   ```
